    90→  create() {
    91→    const data = this.registry.get("initData") as {
    92→      stage?: StageKey;
    93→      save?: SaveData;
    94→    };
    95→    const save = data.save;
    96→    const stageKey: StageKey = save ? (save.stage as StageKey) : data.stage ?? "forest";
    97→    this.stageDef = STAGES[stageKey];
    98→    this.stageW = this.stageDef.width;
    99→    this.stageH = this.stageDef.height;
   100→
   101→    /* ---------- 바닥 ---------- */
   102→    const groundTex = stageKey === "forest" ? "tile_grass" : "tile_dark";
   103→    this.add.tileSprite(0, 0, this.stageW, this.stageH, groundTex).setOrigin(0).setDepth(0);
   104→    // 중앙 가로 길
   105→    this.add
   106→      .tileSprite(0, this.stageH / 2 - 52, this.stageW, 104, "tile_path")
   107→      .setOrigin(0)
   108→      .setDepth(0)
   109→      .setAlpha(0.9);
   110→    if (stageKey === "forest") {
   111→      this.add
   112→        .tileSprite(this.stageW * 0.55 - 52, 0, 104, this.stageH, "tile_path")
   113→        .setOrigin(0)
   114→        .setDepth(0)
   115→        .setAlpha(0.85);
   116→    }
   117→
   118→    this.physics.world.setBounds(0, 0, this.stageW, this.stageH);
   119→    this.cameras.main.setBounds(0, 0, this.stageW, this.stageH);
   120→    this.cameras.main.setBackgroundColor(stageKey === "forest" ? "#0a1408" : "#0d0a1e");
   121→
   122→    // 반응형: 화면 밀도 유지용 카메라 줌 (RESIZE 캔버스 1:1 + 카메라 확대)
   123→    this.applyCameraZoom();
   124→    this.scale.on("resize", this.applyCameraZoom, this);
   125→
   126→    /* ---------- 장식 (F1: 정의된 소수만) ---------- */
   127→    this.placeDecor(stageKey);
   128→
   129→    /* ---------- 플레이어 ---------- */
   130→    const savedPlayer = save;
   131→    this.player = new Player(this, 180, this.stageH / 2);
   132→    if (savedPlayer) {
   133→      this.player.lv = savedPlayer.lv;
   134→      this.player.atk = savedPlayer.atk;
   135→      this.player.maxHp = savedPlayer.maxHp;
   136→      this.player.hp = this.player.maxHp;
   137→    }
   138→    this.playerRef = this.player;
   139→    this.cameras.main.startFollow(this.player, true, 0.12, 0.12);
   140→    this.physics.add.collider(this.player, this.solidGroup);
   141→
   142→    /* ---------- 적 배치 ---------- */
   143→    const rng = new Phaser.Math.RandomDataGenerator([stageKey]);
   144→    for (const group of this.stageDef.enemies) {
   145→      for (let i = 0; i < group.count; i++) {
   146→        let ex = 0;
   147→        let ey = 0;
   148→        let tries = 0;
   149→        do {
   150→          ex = stageKey === "forest" ? rng.between(this.stageW * 0.42, this.stageW - 140) : rng.between(240, this.stageW - 160);
   151→          ey = rng.between(120, this.stageH - 120);
   152→          tries++;
   153→        } while (Phaser.Math.Distance.Between(ex, ey, this.player.x, this.player.y) < 380 && tries < 30);
   154→        const e = new Enemy(this, ex, ey, group.key);
   155→        this.enemies.push(e);
   156→        this.physics.add.collider(e, this.solidGroup);
   157→      }
   158→    }
   159→
   160→    /* ---------- 퀘스트 오브젝트 ---------- */
   161→    if (stageKey === "forest") {
   162→      this.spawnFragment(this.stageW * 0.78, this.stageH * 0.26);
   163→      this.spawnPortal(this.stageW - 130, this.stageH * 0.52);
   164→    } else {
   165→      this.spawnPortalBossArena();
   166→    }
   167→
   168→    /* ---------- 이펙트 풀 ---------- */
   169→    this.buildFxPools();
   170→
   171→    /* ---------- 입력 ---------- */
   172→    this.setupInput();
   173→
   174→    /* ---------- 사운드/BGM ---------- */
   175→    audio.playBGM(stageKey === "forest" ? "field" : "boss");
   176→
   177→    /* ---------- 오프닝 대사 ---------- */
   178→    this.time.delayedCall(400, () => {
   179→      this.showDialogue(stageKey === "forest" ? "intro" : "alfheimIntro");
   180→    });
   181→
   182→    EventBus.emit("ui:playing");
   183→    this.emitHud();
   184→    this.emitQuest();
   185→
   186→    // F2: 거리 실시간 갱신 (300ms 주기 — 프레임 부담 없음)
   187→    this.questTimer = this.time.addEvent({
   188→      delay: 300,
   189→      loop: true,
   190→      callback: () => {
   191→        this.emitQuest();
   192→        this.emitSkills();
   193→      },
   194→    });
   195→
   196→    this.events.once("shutdown", () => this.cleanup());

   207→  private placeDecor(stageKey: StageKey) {
   208→    const def = this.stageDef;
   209→    const rng = new Phaser.Math.RandomDataGenerator([stageKey + "-decor"]);
   210→    this.solidGroup = this.physics.add.staticGroup();
   211→
   212→    // 나무 & 소나무 & 바위 (충돌 있음) — 실제 에셋
   213→    for (let i = 0; i < def.treeCount; i++) {
   214→      const x = rng.between(80, this.stageW - 80);
   215→      const y = rng.between(90, this.stageH - 80);
   216→      if (Math.abs(y - this.stageH / 2) < 90) continue; // 길 위엔 안 심음
   217→      const tex = rng.pick(["tree", "tree", "pine"] as string[]);
   218→      const t = this.add.image(x, y, tex).setDepth(Math.floor(y / 10));
   219→      this.solidGroup.add(t);
   220→      // 64x64 캔버스 하단 줄기 부근만 충돌
   221→      (t.body as Phaser.Physics.Arcade.StaticBody).setSize(20, 14).setOffset(22, 46);
   222→    }
   223→    for (let i = 0; i < def.rockCount; i++) {
   224→      const x = rng.between(80, this.stageW - 80);
   225→      const y = rng.between(90, this.stageH - 80);
   226→      if (Math.abs(y - this.stageH / 2) < 90) continue;
   227→      const r = this.add.image(x, y, "rock").setDepth(Math.floor(y / 10));
   228→      this.solidGroup.add(r);
   229→      (r.body as Phaser.Physics.Arcade.StaticBody).setSize(44, 20).setOffset(10, 40);
   230→    }
   231→
   232→    // F1 핵심: 꽃은 def.flowerCount 송이만 (10 이하)
   233→    const flowers = ["flower_r", "flower_y", "flower_w"];
   234→    for (let i = 0; i < def.flowerCount; i++) {
   235→      const x = rng.between(60, this.stageW - 60);
   236→      const y = rng.between(60, this.stageH - 60);
   237→      this.add.image(x, y, rng.pick(flowers)).setDepth(1).setAlpha(0.95);
   238→    }
   239→
   240→    // 심연 구역(알프헤임) 횃불 — 실제 Kenney 횃불 + 온기 글로우
   241→    if (stageKey === "alfheim") {
   242→      for (let i = 0; i < 6; i++) {
   243→        const tx = 200 + (i * (this.stageW - 400)) / 5;
   244→        const ty = i % 2 === 0 ? this.stageH / 2 - 140 : this.stageH / 2 + 140;
   245→        this.add.image(tx, ty, "torch").setDepth(2);
   246→        const g = this.add
   247→          .image(tx, ty, "glow")
   248→          .setDepth(1)
   249→          .setBlendMode(Phaser.BlendModes.ADD)
   250→          .setTint(0xffa040)
   251→          .setScale(1.1)
   252→          .setAlpha(0.22);
   253→        this.tweens.add({ targets: g, alpha: 0.42, scale: 1.35, duration: 650, yoyo: true, repeat: -1, ease: "Sine.inOut" });
   254→      }
   255→    }
   256→  }

   258→  private spawnFragment(x: number, y: number) {
   259→    this.fragment = this.physics.add.sprite(x, y, "fragment").setDepth(4);
   260→    (this.fragment.body as Phaser.Physics.Arcade.Body).setCircle(13, 3, 3);
   261→
   262→    // 실제 에셋 수정 위 반짝임 (objects.png 별/다이아 프레임)
   263→    this.fragSparkle = this.add
   264→      .sprite(x, y - 14, "sparkle0")
   265→      .setDepth(5)
   266→      .setBlendMode(Phaser.BlendModes.ADD)
   267→      .play("sparkle");
   268→
   269→    // F2 핵심 1: 하늘까지 닿는 빛 기둥 비컨 — 외부 에셋(Kenney Light Masks CC0)
   270→    this.beacon = this.add
   271→      .image(x, y - 128, "beam")
   272→      .setDepth(3)
   273→      .setBlendMode(Phaser.BlendModes.ADD)
   274→      .setAlpha(0.55);
   275→    this.tweens.add({
   276→      targets: this.beacon,
   277→      alpha: { from: 0.4, to: 0.75 },
   278→      scaleX: { from: 1, to: 1.25 },
   279→      duration: 900,
   280→      yoyo: true,
   281→      repeat: -1,
   282→      ease: "Sine.inOut",
   283→    });
   284→    // 바닥 글로우
   285→    const glow = this.add.image(x, y + 8, "glow").setDepth(2).setBlendMode(Phaser.BlendModes.ADD).setScale(1.4);
   286→    this.tweens.add({ targets: glow, scale: 1.7, alpha: 0.7, duration: 800, yoyo: true, repeat: -1 });
   287→    // 뜨는 모션 + 반짝임 동기화
   288→    this.tweens.add({ targets: this.fragment, y: y - 8, duration: 1100, yoyo: true, repeat: -1, ease: "Sine.inOut" });
   289→    if (this.fragSparkle) {
   290→      this.tweens.add({ targets: this.fragSparkle, y: y - 22, duration: 1100, yoyo: true, repeat: -1, ease: "Sine.inOut" });
   291→    }
   292→
   293→    this.physics.add.overlap(this.player, this.fragment, () => {
   294→      if (!this.fragment || !this.fragment.active) return;
   295→      this.collectFragment(glow);
   296→    });
   297→  }

   299→  private collectFragment(glow: Phaser.GameObjects.Image) {
   300→    audio.sfx.pickup();
   301→    audio.sfx.questDone();
   302→    this.player.atk += 5; // 파편 보너스
   303→    this.player.healFull();
   304→    glow.destroy();
   305→    this.tweens.add({ targets: this.beacon, alpha: 0, duration: 400, onComplete: () => this.beacon?.destroy() });
   306→    this.beacon = null;
   307→    this.fragSparkle?.destroy();
   308→    this.fragSparkle = null;
   309→    const f = this.fragment!;
   310→    this.tweens.add({
   311→      targets: f,
   312→      y: f.y - 60,
   313→      alpha: 0,
   314→      scale: 1.6,
   315→      duration: 500,
   316→      onComplete: () => f.destroy(),
   317→    });
   318→    this.fragment = null;
   319→    this.spawnBurstAt(f.x, f.y, 14, 0x9df0ff);
   320→    this.showDialogue("fragment");
   321→    this.advanceQuest();
   322→    // 늑대를 먼저 다 잡아둔 경우 — 토벌 퀘스트가 이미 충족됐으면 즉시 완료 처리
   323→    this.time.delayedCall(100, () => {
   324→      if (this.stageDef.key === "forest") this.tryCompleteHunt("wolf");
   325→      else this.tryCompleteHunt("minion");
   326→    });
   327→    this.save();
   328→  }

   330→  private spawnPortal(x: number, y: number) {
   331→    // 외부 에셋 차원문(varkalandar CC-BY, 8프레임 소용돌이) — 비활성 시 회색 틴트
   332→    this.portal = this.physics.add.sprite(x, y, "portal0").setDepth(3).setTint(0x777777);
   333→    (this.portal.body as Phaser.Physics.Arcade.Body).setSize(34, 44).setOffset(15, 10);
   334→    this.physics.add.overlap(this.player, this.portal, () => {
   335→      if (!this.portalActive || !this.portal?.active) return;
   336→      this.enterPortal();
   337→    });

   340→  private activatePortal() {
   341→    if (!this.portal) return;
   342→    this.portalActive = true;
   343→    this.portal.clearTint();
   344→    this.portal.play("portal-spin");
   345→    audio.sfx.portal();
   346→    // F2: 차원문에도 작은 비컨
   347→    this.portalBeacon = this.add
   348→      .image(this.portal.x, this.portal.y - 120, "beam")
   349→      .setDepth(2)
   350→      .setBlendMode(Phaser.BlendModes.ADD)
   351→      .setAlpha(0.4)
   352→      .setTint(0x9d7aff);
   353→    this.tweens.add({
   354→      targets: this.portalBeacon,
   355→      alpha: { from: 0.28, to: 0.5 },
   356→      duration: 800,
   357→      yoyo: true,
   358→      repeat: -1,
   359→    });
   360→    this.showBanner("차원문이 열렸다!");
   361→  }

   363→  private enterPortal() {
   364→    this.portalActive = false;
   365→    audio.sfx.portal();
   366→    this.cameras.main.fadeOut(500, 0, 0, 0);
   367→    this.player.state = "idle";
   368→    this.time.delayedCall(520, () => {
   369→      this.saveStage("alfheim");
   370→      this.scene.restart({ stage: "alfheim" });
   371→    });
   372→  }

   380→  private buildFxPools() {
   381→    // ⚠️ 씬 재시작(스테이지 전환) 시 풀이 누적되지 않도록 리셋
   382→    //   (파괴된 텍스트 재사용 → setText 크래시 원인)
   383→    for (const d of this.dmgPool) d.destroy();
   384→    this.dmgPool = [];
   385→    for (const s of this.slashPool) s.destroy();
   386→    this.slashPool = [];
   387→    for (const s of this.starPool) s.destroy();
   388→    this.starPool = [];
   389→
   390→    // 공유 파티클 이미터 2종
   391→    this.hitEmitter = this.add.particles(0, 0, "spark", {
   392→      lifespan: 260,
   393→      speed: { min: 60, max: 190 },
   394→      scale: { start: 1, end: 0 },
   395→      emitting: false,
   396→      blendMode: Phaser.BlendModes.ADD,
   397→    }).setDepth(30);
   398→    this.burstEmitter = this.add.particles(0, 0, "spark", {
   399→      lifespan: 480,
   400→      speed: { min: 90, max: 300 },
   401→      scale: { start: 1.4, end: 0 },
   402→      emitting: false,
   403→      blendMode: Phaser.BlendModes.ADD,
   404→    }).setDepth(30);
   405→
   406→    // 데미지 텍스트 12장 고정 풀
   407→    for (let i = 0; i < 12; i++) {
   408→      const t = this.add
   409→        .text(0, 0, "", {
   410→          fontFamily: "sans-serif",
   411→          fontSize: "17px",
   412→          color: "#ffffff",
   413→          stroke: "#1a1020",
   414→          strokeThickness: 4,
   415→          fontStyle: "bold",
   416→        })
   417→        .setDepth(40)
   418→        .setActive(false)
   419→        .setVisible(false);
   420→      this.dmgPool.push(t);
   421→    }
   422→
   423→    // 참격(초승달 애니) 스프라이트 5장 고정 풀 — 외부 에셋(Cethiel CC0)
   424→    for (let i = 0; i < 5; i++) {
   425→      const s = this.add
   426→        .sprite(0, 0, "slash0")
   427→        .setDepth(25)
   428→        .setBlendMode(Phaser.BlendModes.ADD)
   429→        .setActive(false)
   430→        .setVisible(false);
   431→      this.slashPool.push(s);
   432→    }
   433→
   434→    // 타격 스타 4장 고정 풀 (실제 에셋: objects.png 폭발 프레임)
   435→    for (let i = 0; i < 4; i++) {
   436→      const s = this.add
   437→        .image(0, 0, "impact_star")
   438→        .setDepth(29)
   439→        .setBlendMode(Phaser.BlendModes.ADD)
   440→        .setActive(false)
   441→        .setVisible(false);
   442→      this.starPool.push(s);
   443→    }
   444→  }

   446→  spawnDamageText(x: number, y: number, val: number) {

   746→  update(_time: number, delta: number) {
   747→    const dt = Math.min(delta, 50);
   748→    if (this.dialoguing || !this.player) return;
   749→    if (this.player.state === "dead") return;
   750→
   751→    // 키보드 이동
   752→    const mv = new Phaser.Math.Vector2(0, 0);
   753→    if (this.keys.A.isDown || this.keys.LEFT.isDown) mv.x -= 1;
   754→    if (this.keys.D.isDown || this.keys.RIGHT.isDown) mv.x += 1;
   755→    if (this.keys.W.isDown || this.keys.UP.isDown) mv.y -= 1;
   756→    if (this.keys.S.isDown || this.keys.DOWN.isDown) mv.y += 1;
   757→    if (mv.lengthSq() > 0) mv.normalize();
   758→
   759→    // 터치 우선
   760→    const useTouch = this.touchMove.lengthSq() > 0.01;
   761→    const move = useTouch ? this.touchMove : mv;
   762→
   763→    // 키보드 공격/스킬
   764→    if (Phaser.Input.Keyboard.JustDown(this.keys.SPACE) || Phaser.Input.Keyboard.JustDown(this.keys.J))
   765→      this.attackQueued = true;
   766→    if (Phaser.Input.Keyboard.JustDown(this.keys.K)) this.player.useSkill1();
   767→    if (Phaser.Input.Keyboard.JustDown(this.keys.L)) this.player.useSkill2();
   768→
   769→    this.player.update(dt, move, this.attackQueued);
   770→    this.attackQueued = false;
   771→
   772→    // 적 AI
   773→    for (const e of this.enemies) {
   774→      if (e.active && e.alive) e.tick(dt, this.player);
   775→    }
   776→    // 보스 AI
   777→    this.boss?.tick(dt, this.player);
   778→
   779→    // F2 핵심 2: 화면 가장자리 화살표 — 목표물이 안 보일 때 방향 안내
   780→    this.updateEdgeArrow();
   781→  }

   796→  private questTargetPos(): Phaser.Math.Vector2 | null {
   797→    const q = this.currentQuest();
   798→    if (!q) return null;
   799→    switch (q.type) {
   800→      case "collect":
   801→        return this.fragment && this.fragment.active ? new Phaser.Math.Vector2(this.fragment.x, this.fragment.y) : null;
   802→      case "hunt": {
   803→        let best: Enemy | null = null;
   804→        let bd = Infinity;
   805→        for (const e of this.enemies) {
   806→          if (!e.active || !e.alive) continue;
   807→          const d = Phaser.Math.Distance.Between(e.x, e.y, this.player.x, this.player.y);
   808→          if (d < bd) {
   809→            bd = d;
   810→            best = e;
   811→          }
   812→        }
   813→        return best ? new Phaser.Math.Vector2(best.x, best.y) : null;
   814→      }
   815→      case "reach":
   816→        return this.portal && this.portal.active ? new Phaser.Math.Vector2(this.portal.x, this.portal.y) : null;
   817→      case "boss":
   818→        return this.boss && this.boss.active && this.boss.alive
   819→          ? new Phaser.Math.Vector2(this.boss.x, this.boss.y)
   820→          : null;
   821→      default:
   822→        return null;
   823→    }
   824→  }

   826→  private updateEdgeArrow() {
   827→    const target = this.questTargetPos();
   828→    if (!target) {
   829→      if (this.edgeArrow) {
   830→        this.edgeArrow.destroy();
   831→        this.edgeArrow = null;
   832→      }
   833→      if (this.questMark) {
   834→        this.questMark.destroy();
   835→        this.questMark = null;
   836→      }
   837→      return;
   838→    }
   839→
   840→    // 목표물 바로 위 퀘스트 마커(?) — 외부 에셋(Zelda-like CC0 말풍선)
   841→    if (!this.questMark) {
   842→      this.questMark = this.add.image(target.x, target.y - 34, "quest_mark").setDepth(22).setScale(1.3);
   843→    }
   844→    this.questMark.setPosition(target.x, target.y - 34 + Math.sin(this.time.now / 200) * 3);
   845→
   846→    // 카메라 뷰 밖이면 가장자리 화살표 표시
   847→    const view = this.cameras.main.worldView;
   848→    const margin = 40;
   849→    const inside =
   850→      target.x > view.x + margin &&
   851→      target.x < view.right - margin &&
   852→      target.y > view.y + margin &&
   853→      target.y < view.bottom - margin;
   854→
   855→    if (!inside) {
   856→      if (!this.edgeArrow)
   857→        this.edgeArrow = this.add
   858→          .image(0, 0, "edge_arrow")
   859→          .setDepth(50)
   860→          .setScrollFactor(0)
   861→          .setTint(0xffd76a)
   862→          .setBlendMode(Phaser.BlendModes.ADD);
   863→      const cx = this.cameras.main.width / 2;
   864→      const cy = this.cameras.main.height / 2;
   865→      const angle = Phaser.Math.Angle.Between(cx, cy, target.x - view.x, target.y - view.y);
   866→      // 화면 중심에서 끝까지 레이캐스트하여 클램프
   867→      const halfW = this.cameras.main.width / 2 - 46;
   868→      const halfH = this.cameras.main.height / 2 - 46;
   869→      const t = Math.min(
   870→        Math.abs(halfW / Math.cos(angle)) || Infinity,
   871→        Math.abs(halfH / Math.sin(angle)) || Infinity
   872→      );
   873→      this.edgeArrow.setPosition(cx + Math.cos(angle) * t, cy + Math.sin(angle) * t);
   874→      this.edgeArrow.setRotation(angle);
   875→      this.edgeArrow.setAlpha(0.65 + Math.sin(this.time.now / 150) * 0.3);
   876→    } else if (this.edgeArrow) {
   877→      this.edgeArrow.destroy();
   878→      this.edgeArrow = null;
   879→    }
   880→  }

   901→        target: 1,
   902→        distance: null,
   903→      } satisfies QuestState);
   904→      return;
   905→    }
   906→    let current = 0;
   907→    let target = 1;
   908→    if (q.type === "hunt") {
   909→      current = Math.min(this.huntCount, q.need ?? 0);
   910→      target = q.need ?? 0;
   911→    }
   912→    let distance: number | null = null;
   913→    const t = this.questTargetPos();
   914→    if (t) distance = Math.round(Phaser.Math.Distance.Between(this.player.x, this.player.y, t.x, t.y) / 32);
   915→    EventBus.emit("quest", {
   916→      title: q.title,
   917→      desc: q.desc,
   918→      current,
   919→      target,
   920→      distance,
   921→    } satisfies QuestState);
   922→  }
   923→

   926→    EventBus.emit("skills", {

   929→      s1Max: this.player.skill1Max,
