   207→   207→  private placeDecor(stageKey: StageKey) {
   208→   208→    const def = this.stageDef;
   209→   209→    const rng = new Phaser.Math.RandomDataGenerator([stageKey + "-decor"]);
   210→   210→    this.solidGroup = this.physics.add.staticGroup();
   211→   211→
   212→   212→    // 나무 & 소나무 & 바위 (충돌 있음) — 실제 에셋
   213→   213→    for (let i = 0; i < def.treeCount; i++) {
   214→   214→      const x = rng.between(80, this.stageW - 80);
   215→   215→      const y = rng.between(90, this.stageH - 80);
   216→   216→      if (Math.abs(y - this.stageH / 2) < 90) continue; // 길 위엔 안 심음
   217→   217→      const tex = rng.pick(["tree", "tree", "pine"] as string[]);
   218→   218→      const t = this.add.image(x, y, tex).setDepth(Math.floor(y / 10));
   219→   219→      this.solidGroup.add(t);
   220→   220→      // 64x64 캔버스 하단 줄기 부근만 충돌
   221→   221→      (t.body as Phaser.Physics.Arcade.StaticBody).setSize(20, 14).setOffset(22, 46);
   222→   222→    }
   223→   223→    for (let i = 0; i < def.rockCount; i++) {
   224→   224→      const x = rng.between(80, this.stageW - 80);
   225→   225→      const y = rng.between(90, this.stageH - 80);
   226→   226→      if (Math.abs(y - this.stageH / 2) < 90) continue;
   227→   227→      const r = this.add.image(x, y, "rock").setDepth(Math.floor(y / 10));
   228→   228→      this.solidGroup.add(r);
   229→   229→      (r.body as Phaser.Physics.Arcade.StaticBody).setSize(44, 20).setOffset(10, 40);
   230→   230→    }
   231→   231→
   232→   232→    // F1 핵심: 꽃은 def.flowerCount 송이만 (10 이하)
   233→   233→    const flowers = ["flower_r", "flower_y", "flower_w"];
   234→   234→    for (let i = 0; i < def.flowerCount; i++) {
   235→   235→      const x = rng.between(60, this.stageW - 60);
   236→   236→      const y = rng.between(60, this.stageH - 60);
   237→   237→      this.add.image(x, y, rng.pick(flowers)).setDepth(1).setAlpha(0.95);
   238→   238→    }
   239→   239→
   240→   240→    // 심연 구역(알프헤임) 횃불 — 실제 Kenney 횃불 + 온기 글로우
   241→   241→    if (stageKey === "alfheim") {
   242→   242→      for (let i = 0; i < 6; i++) {
   243→   243→        const tx = 200 + (i * (this.stageW - 400)) / 5;
   244→   244→        const ty = i % 2 === 0 ? this.stageH / 2 - 140 : this.stageH / 2 + 140;
   245→   245→        this.add.image(tx, ty, "torch").setDepth(2);
   246→   246→        const g = this.add
   247→   247→          .image(tx, ty, "glow")
   248→   248→          .setDepth(1)
   249→   249→          .setBlendMode(Phaser.BlendModes.ADD)
   250→   250→          .setTint(0xffa040)
   251→   251→          .setScale(1.1)
   252→   252→          .setAlpha(0.22);
   253→   253→        this.tweens.add({ targets: g, alpha: 0.42, scale: 1.35, duration: 650, yoyo: true, repeat: -1, ease: "Sine.inOut" });
   254→   254→      }
   255→   255→    }
   256→   256→  }

   746→   746→  update(_time: number, delta: number) {
   747→   747→    const dt = Math.min(delta, 50);
   748→   748→    if (this.dialoguing || !this.player) return;
   749→   749→    if (this.player.state === "dead") return;
   750→   750→
   751→   751→    // 키보드 이동
   752→   752→    const mv = new Phaser.Math.Vector2(0, 0);
   753→   753→    if (this.keys.A.isDown || this.keys.LEFT.isDown) mv.x -= 1;
   754→   754→    if (this.keys.D.isDown || this.keys.RIGHT.isDown) mv.x += 1;
   755→   755→    if (this.keys.W.isDown || this.keys.UP.isDown) mv.y -= 1;
   756→   756→    if (this.keys.S.isDown || this.keys.DOWN.isDown) mv.y += 1;
   757→   757→    if (mv.lengthSq() > 0) mv.normalize();
   758→   758→
   759→   759→    // 터치 우선
   760→   760→    const useTouch = this.touchMove.lengthSq() > 0.01;
   761→   761→    const move = useTouch ? this.touchMove : mv;
   762→   762→
   763→   763→    // 키보드 공격/스킬
   764→   764→    if (Phaser.Input.Keyboard.JustDown(this.keys.SPACE) || Phaser.Input.Keyboard.JustDown(this.keys.J))
   765→   765→      this.attackQueued = true;
   766→   766→    if (Phaser.Input.Keyboard.JustDown(this.keys.K)) this.player.useSkill1();
   767→   767→    if (Phaser.Input.Keyboard.JustDown(this.keys.L)) this.player.useSkill2();
   768→   768→
   769→   769→    this.player.update(dt, move, this.attackQueued);
   770→   770→    this.attackQueued = false;
   771→   771→
   772→   772→    // 적 AI
   773→   773→    for (const e of this.enemies) {
   774→   774→      if (e.active && e.alive) e.tick(dt, this.player);
   775→   775→    }
   776→   776→    // 보스 AI
   777→   777→    this.boss?.tick(dt, this.player);
   778→   778→
   779→   779→    // F2 핵심 2: 화면 가장자리 화살표 — 목표물이 안 보일 때 방향 안내
   780→   780→    this.updateEdgeArrow();
   781→   781→  }

   784→   784→    return this.touchMove.lengthSq() > 0.01 ? this.touchMove.clone() : new Phaser.Math.Vector2();
   785→   785→  }

   787→   787→  getAllTargets(): (Enemy | Boss)[] {
   788→   788→    const list: (Enemy | Boss)[] = [];
   789→   789→    for (const e of this.enemies) if (e.active && e.alive) list.push(e);
   790→   790→    if (this.boss && this.boss.active && this.boss.alive) list.push(this.boss);

   796→   796→  private questTargetPos(): Phaser.Math.Vector2 | null {
   797→   797→    const q = this.currentQuest();
   798→   798→    if (!q) return null;
   799→   799→    switch (q.type) {
   800→   800→      case "collect":
   801→   801→        return this.fragment && this.fragment.active ? new Phaser.Math.Vector2(this.fragment.x, this.fragment.y) : null;
   802→   802→      case "hunt": {
   803→   803→        let best: Enemy | null = null;
   804→   804→        let bd = Infinity;
   805→   805→        for (const e of this.enemies) {
   806→   806→          if (!e.active || !e.alive) continue;
   807→   807→          const d = Phaser.Math.Distance.Between(e.x, e.y, this.player.x, this.player.y);
   808→   808→          if (d < bd) {
   809→   809→            bd = d;
   810→   810→            best = e;
   811→   811→          }
   812→   812→        }
   813→   813→        return best ? new Phaser.Math.Vector2(best.x, best.y) : null;
   814→   814→      }
   815→   815→      case "reach":
   816→   816→        return this.portal && this.portal.active ? new Phaser.Math.Vector2(this.portal.x, this.portal.y) : null;
   817→   817→      case "boss":
   818→   818→        return this.boss && this.boss.active && this.boss.alive
   819→   819→          ? new Phaser.Math.Vector2(this.boss.x, this.boss.y)
   820→   820→          : null;
   821→   821→      default:
   822→   822→        return null;
   823→   823→    }
   824→   824→  }

   826→   826→  private updateEdgeArrow() {
   827→   827→    const target = this.questTargetPos();
   828→   828→    if (!target) {
   829→   829→      if (this.edgeArrow) {
   830→   830→        this.edgeArrow.destroy();
   831→   831→        this.edgeArrow = null;
   832→   832→      }
   833→   833→      if (this.questMark) {
   834→   834→        this.questMark.destroy();
   835→   835→        this.questMark = null;
   836→   836→      }
   837→   837→      return;
   838→   838→    }
   839→   839→
   840→   840→    // 목표물 바로 위 퀘스트 마커(?) — 외부 에셋(Zelda-like CC0 말풍선)
   841→   841→    if (!this.questMark) {
   842→   842→      this.questMark = this.add.image(target.x, target.y - 34, "quest_mark").setDepth(22).setScale(1.3);
   843→   843→    }
   844→   844→    this.questMark.setPosition(target.x, target.y - 34 + Math.sin(this.time.now / 200) * 3);
   845→   845→
   846→   846→    // 카메라 뷰 밖이면 가장자리 화살표 표시
   847→   847→    const view = this.cameras.main.worldView;
   848→   848→    const margin = 40;
   849→   849→    const inside =
   850→   850→      target.x > view.x + margin &&
   851→   851→      target.x < view.right - margin &&
   852→   852→      target.y > view.y + margin &&
   853→   853→      target.y < view.bottom - margin;
   854→   854→
   855→   855→    if (!inside) {
   856→   856→      if (!this.edgeArrow)
   857→   857→        this.edgeArrow = this.add
   858→   858→          .image(0, 0, "edge_arrow")
   859→   859→          .setDepth(50)
   860→   860→          .setScrollFactor(0)
   861→   861→          .setTint(0xffd76a)
   862→   862→          .setBlendMode(Phaser.BlendModes.ADD);
   863→   863→      const cx = this.cameras.main.width / 2;
   864→   864→      const cy = this.cameras.main.height / 2;
   865→   865→      const angle = Phaser.Math.Angle.Between(cx, cy, target.x - view.x, target.y - view.y);
   866→   866→      // 화면 중심에서 끝까지 레이캐스트하여 클램프
   867→   867→      const halfW = this.cameras.main.width / 2 - 46;
   868→   868→      const halfH = this.cameras.main.height / 2 - 46;
   869→   869→      const t = Math.min(
   870→   870→        Math.abs(halfW / Math.cos(angle)) || Infinity,
   871→   871→        Math.abs(halfH / Math.sin(angle)) || Infinity
   872→   872→      );
   873→   873→      this.edgeArrow.setPosition(cx + Math.cos(angle) * t, cy + Math.sin(angle) * t);
   874→   874→      this.edgeArrow.setRotation(angle);
   875→   875→      this.edgeArrow.setAlpha(0.65 + Math.sin(this.time.now / 150) * 0.3);
   876→   876→    } else if (this.edgeArrow) {
   877→   877→      this.edgeArrow.destroy();
   878→   878→      this.edgeArrow = null;
   879→   879→    }
   880→   880→  }

   894→   894→  emitQuest() {
   895→   895→    const q = this.currentQuest();
   896→   896→    if (!q) {
   897→   897→      EventBus.emit("quest", {
   898→   898→        title: "지역 클리어!",
   899→   899→        desc: "",
   900→   900→        current: 1,
   901→   901→        target: 1,
   902→   902→        distance: null,
   903→   903→      } satisfies QuestState);
   904→   904→      return;
   905→   905→    }
   906→   906→    let current = 0;
   907→   907→    let target = 1;
   908→   908→    if (q.type === "hunt") {
   909→   909→      current = Math.min(this.huntCount, q.need ?? 0);
   910→   910→      target = q.need ?? 0;
   911→   911→    }
   912→   912→    let distance: number | null = null;
   913→   913→    const t = this.questTargetPos();
   914→   914→    if (t) distance = Math.round(Phaser.Math.Distance.Between(this.player.x, this.player.y, t.x, t.y) / 32);
   915→   915→    EventBus.emit("quest", {
   916→   916→      title: q.title,
   917→   917→      desc: q.desc,
   918→   918→      current,
   919→   919→      target,
   920→   920→      distance,
   921→   921→    } satisfies QuestState);
   922→   922→  }
