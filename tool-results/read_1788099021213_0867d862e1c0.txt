    43→/** 강화 비용 — 단계별 눈덩이 곡선 (높은 단계일수록 골드 소모 급증 → 골드 싱크) */
    44→export function upgradeCost(slot: "weapon" | "armor", level: number): number {
    45→  const base = slot === "weapon" ? 45 : 38;
    46→  return Math.round(base * Math.pow(1.45, level));
    47→}

   846→/* ================= 스탯 자동 배분 (v2.0 — 사용자 지시 #18, 메이플 4:1 감각) ================= */
   847→
   848→export const AUTO_ALLOC: Record<FamilyKey, { str: number; dex: number; int: number; luk: number }> = {
   849→  warrior: { str: 4, dex: 1, int: 0, luk: 0 }, // 힘 4 : 민첩 1
   850→  ranger: { str: 1, dex: 4, int: 0, luk: 0 }, // 민첩 4 : 힘 1
   851→  mage: { str: 0, dex: 0, int: 4, luk: 1 }, // 지력 4 : 행운 1
   852→};

   854→/** AP를 계열 배분 비율로 나눠 담는다 (남는 AP는 주스탯에 몰아줌) */
   855→export function autoAllocPlan(family: FamilyKey, ap: number): { str: number; dex: number; int: number; luk: number } {
   856→  const r = AUTO_ALLOC[family];
   857→  const total = r.str + r.dex + r.int + r.luk;
   858→  const out = { str: 0, dex: 0, int: 0, luk: 0 };
   859→  if (total === 0 || ap <= 0) return out;
   860→  const keys: (keyof typeof out)[] = ["str", "dex", "int", "luk"];
   861→  let left = ap;
   862→  // 라운드 로빈 배분 — 4:1 비율 유지
   863→  let round = 0;
   864→  while (left > 0) {
   865→    for (const k of keys) {
   866→      if (left <= 0) break;
   867→      if (r[k] > 0 && round < r[k]) {
   868→        out[k]++;
   869→        left--;
   870→      }
   871→    }
   872→    if (keys.every((k) => r[k] === 0 || round >= r[k])) {
   873→      // 비율 1사이클 완료 → 주스탯 몰아주기
   874→      const main = keys.reduce((a, b) => (r[a] >= r[b] ? a : b));
   875→      out[main] += left;
   876→      left = 0;
   877→    }
   878→    round++;
   879→  }
   880→  return out;
   881→}
