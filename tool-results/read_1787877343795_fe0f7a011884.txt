    81→    81→  create() {
    82→    82→    const data = this.registry.get("initData") as {
    83→    83→      stage?: StageKey;
    84→    84→      save?: SaveData;
    85→    85→    };
    86→    86→    const save = data.save;
    87→    87→    const stageKey: StageKey = save ? (save.stage as StageKey) : data.stage ?? "forest";
    88→    88→    this.stageDef = STAGES[stageKey];
    89→    89→    this.stageW = this.stageDef.width;
    90→    90→    this.stageH = this.stageDef.height;
    91→    91→
    92→    92→    /* ---------- 바닥 ---------- */
    93→    93→    const groundTex = stageKey === "forest" ? "tile_grass" : "tile_dark";
    94→    94→    this.add.tileSprite(0, 0, this.stageW, this.stageH, groundTex).setOrigin(0).setDepth(0);
    95→    95→    // 중앙 가로 길
    96→    96→    this.add
    97→    97→      .tileSprite(0, this.stageH / 2 - 52, this.stageW, 104, "tile_path")
    98→    98→      .setOrigin(0)
    99→    99→      .setDepth(0)
   100→   100→      .setAlpha(0.9);
   101→   101→    if (stageKey === "forest") {
   102→   102→      this.add
   103→   103→        .tileSprite(this.stageW * 0.55 - 52, 0, 104, this.stageH, "tile_path")
   104→   104→        .setOrigin(0)
   105→   105→        .setDepth(0)
   106→   106→        .setAlpha(0.85);
   107→   107→    }
   108→   108→
   109→   109→    this.physics.world.setBounds(0, 0, this.stageW, this.stageH);
   110→   110→    this.cameras.main.setBounds(0, 0, this.stageW, this.stageH);
   111→   111→    this.cameras.main.setBackgroundColor(stageKey === "forest" ? "#0a1408" : "#0d0a1e");
   112→   112→
   113→   113→    /* ---------- 장식 (F1: 정의된 소수만) ---------- */
   114→   114→    this.placeDecor(stageKey);
   115→   115→
   116→   116→    /* ---------- 플레이어 ---------- */
   117→   117→    const savedPlayer = save;
   118→   118→    this.player = new Player(this, 180, this.stageH / 2);
   119→   119→    if (savedPlayer) {
   120→   120→      this.player.lv = savedPlayer.lv;
   121→   121→      this.player.atk = savedPlayer.atk;
   122→   122→      this.player.maxHp = savedPlayer.maxHp;
   123→   123→      this.player.hp = this.player.maxHp;
   124→   124→    }
   125→   125→    this.playerRef = this.player;
   126→   126→    this.cameras.main.startFollow(this.player, true, 0.12, 0.12);
   127→   127→    this.physics.add.collider(this.player, this.solidGroup);
   128→   128→
   129→   129→    /* ---------- 적 배치 ---------- */
   130→   130→    const rng = new Phaser.Math.RandomDataGenerator([stageKey]);
   131→   131→    for (const group of this.stageDef.enemies) {
   132→   132→      for (let i = 0; i < group.count; i++) {
   133→   133→        let ex = 0;
   134→   134→        let ey = 0;
   135→   135→        let tries = 0;
   136→   136→        do {
   137→   137→          ex = stageKey === "forest" ? rng.between(this.stageW * 0.42, this.stageW - 140) : rng.between(240, this.stageW - 160);
   138→   138→          ey = rng.between(120, this.stageH - 120);
   139→   139→          tries++;
   140→   140→        } while (Phaser.Math.Distance.Between(ex, ey, this.player.x, this.player.y) < 380 && tries < 30);
   141→   141→        const e = new Enemy(this, ex, ey, group.key);
   142→   142→        this.enemies.push(e);
   143→   143→        this.physics.add.collider(e, this.solidGroup);
   144→   144→      }
   145→   145→    }
   146→   146→
   147→   147→    /* ---------- 퀘스트 오브젝트 ---------- */
   148→   148→    if (stageKey === "forest") {
   149→   149→      this.spawnFragment(this.stageW * 0.78, this.stageH * 0.26);
   150→   150→      this.spawnPortal(this.stageW - 130, this.stageH * 0.52);
   151→   151→    } else {
   152→   152→      this.spawnPortalBossArena();
   153→   153→    }
   154→   154→
   155→   155→    /* ---------- 이펙트 풀 ---------- */
   156→   156→    this.buildFxPools();
   157→   157→
   158→   158→    /* ---------- 입력 ---------- */
   159→   159→    this.setupInput();
   160→   160→
   161→   161→    /* ---------- 사운드/BGM ---------- */
   162→   162→    audio.playBGM(stageKey === "forest" ? "field" : "boss");
   163→   163→
   164→   164→    /* ---------- 오프닝 대사 ---------- */
   165→   165→    this.time.delayedCall(400, () => {
   166→   166→      this.showDialogue(stageKey === "forest" ? "intro" : "alfheimIntro");
   167→   167→    });
   168→   168→
   169→   169→    EventBus.emit("ui:playing");
   170→   170→    this.emitHud();
   171→   171→    this.emitQuest();
   172→   172→
   173→   173→    // F2: 거리 실시간 갱신 (300ms 주기 — 프레임 부담 없음)
   174→   174→    this.questTimer = this.time.addEvent({
   175→   175→      delay: 300,
   176→   176→      loop: true,
   177→   177→      callback: () => {
   178→   178→        this.emitQuest();
   179→   179→        this.emitSkills();
   180→   180→      },
   181→   181→    });
   182→   182→
   183→   183→    this.events.once("shutdown", () => this.cleanup());
   184→   184→  }

   190→   190→  private placeDecor(stageKey: StageKey) {
   191→   191→    const def = this.stageDef;
   192→   192→    const rng = new Phaser.Math.RandomDataGenerator([stageKey + "-decor"]);
   193→   193→    this.solidGroup = this.physics.add.staticGroup();
   194→   194→
   195→   195→    // 나무 & 바위 (충돌 있음)
   196→   196→    for (let i = 0; i < def.treeCount; i++) {
   197→   197→      const x = rng.between(80, this.stageW - 80);
   198→   198→      const y = rng.between(90, this.stageH - 80);
   199→   199→      if (Math.abs(y - this.stageH / 2) < 90) continue; // 길 위엔 안 심음
   200→   200→      const t = this.add.image(x, y, "tree").setDepth(Math.floor(y / 10));
   201→   201→      this.solidGroup.add(t);
   202→   202→      (t.body as Phaser.Physics.Arcade.StaticBody).setSize(24, 16).setOffset(16, 66);
   203→   203→    }
   204→   204→    for (let i = 0; i < def.rockCount; i++) {
   205→   205→      const x = rng.between(80, this.stageW - 80);
   206→   206→      const y = rng.between(90, this.stageH - 80);
   207→   207→      if (Math.abs(y - this.stageH / 2) < 90) continue;
   208→   208→      const r = this.add.image(x, y, "rock").setDepth(Math.floor(y / 10));
   209→   209→      this.solidGroup.add(r);
   210→   210→      (r.body as Phaser.Physics.Arcade.StaticBody).setSize(30, 14).setOffset(3, 10);
   211→   211→    }
   212→   212→
   213→   213→    // F1 핵심: 꽃은 def.flowerCount 송이만 (10 이하)
   214→   214→    const flowers = ["flower_r", "flower_y", "flower_w"];
   215→   215→    for (let i = 0; i < def.flowerCount; i++) {
   216→   216→      const x = rng.between(60, this.stageW - 60);
   217→   217→      const y = rng.between(60, this.stageH - 60);
   218→   218→      this.add.image(x, y, rng.pick(flowers)).setDepth(1).setAlpha(0.95);
   219→   219→    }
   220→   220→  }

   222→   222→  private spawnFragment(x: number, y: number) {
   223→   223→    this.fragment = this.physics.add.sprite(x, y, "fragment").setDepth(4);
   224→   224→    (this.fragment.body as Phaser.Physics.Arcade.Body).setCircle(12, -3, -2);
   225→   225→
   226→   226→    // F2 핵심 1: 하늘까지 닿는 빛 기둥 비컨 — 멀리서도 확실히 보임
   227→   227→    this.beacon = this.add
   228→   228→      .image(x, y - 240, "beacon")
   229→   229→      .setDepth(3)
   230→   230→      .setBlendMode(Phaser.BlendModes.ADD)
   231→   231→      .setAlpha(0.55);
   232→   232→    this.tweens.add({
   233→   233→      targets: this.beacon,
   234→   234→      alpha: { from: 0.4, to: 0.75 },
   235→   235→      scaleX: { from: 1, to: 1.25 },
   236→   236→      duration: 900,
   237→   237→      yoyo: true,
   238→   238→      repeat: -1,
   239→   239→      ease: "Sine.inOut",
   240→   240→    });
   241→   241→    // 바닥 글로우
   242→   242→    const glow = this.add.image(x, y + 8, "glow").setDepth(2).setBlendMode(Phaser.BlendModes.ADD).setScale(1.4);
   243→   243→    this.tweens.add({ targets: glow, scale: 1.7, alpha: 0.7, duration: 800, yoyo: true, repeat: -1 });
   244→   244→    // 뜨는 모션
   245→   245→    this.tweens.add({ targets: this.fragment, y: y - 8, duration: 1100, yoyo: true, repeat: -1, ease: "Sine.inOut" });
   246→   246→
   247→   247→    this.physics.add.overlap(this.player, this.fragment, () => {
   248→   248→      if (!this.fragment || !this.fragment.active) return;
   249→   249→      this.collectFragment(glow);
   250→   250→    });
   251→   251→  }

   253→   253→  private collectFragment(glow: Phaser.GameObjects.Image) {

   262→   262→    this.tweens.add({
   263→   263→      targets: f,

   265→   265→      alpha: 0,
   266→   266→      scale: 1.6,
   267→   267→      duration: 500,
   268→   268→      onComplete: () => f.destroy(),
   269→   269→    });
   270→   270→    this.fragment = null;

   595→   595→  update(_time: number, delta: number) {
   596→   596→    const dt = Math.min(delta, 50);
   597→   597→    if (this.dialoguing || !this.player) return;
   598→   598→    if (this.player.state === "dead") return;
   599→   599→
   600→   600→    // 키보드 이동
   601→   601→    const mv = new Phaser.Math.Vector2(0, 0);
   602→   602→    if (this.keys.A.isDown || this.keys.LEFT.isDown) mv.x -= 1;
   603→   603→    if (this.keys.D.isDown || this.keys.RIGHT.isDown) mv.x += 1;
   604→   604→    if (this.keys.W.isDown || this.keys.UP.isDown) mv.y -= 1;
   605→   605→    if (this.keys.S.isDown || this.keys.DOWN.isDown) mv.y += 1;
   606→   606→    if (mv.lengthSq() > 0) mv.normalize();
   607→   607→
   608→   608→    // 터치 우선
   609→   609→    const useTouch = this.touchMove.lengthSq() > 0.01;
   610→   610→    const move = useTouch ? this.touchMove : mv;
   611→   611→
   612→   612→    // 키보드 공격/스킬
   613→   613→    if (Phaser.Input.Keyboard.JustDown(this.keys.SPACE) || Phaser.Input.Keyboard.JustDown(this.keys.J))
   614→   614→      this.attackQueued = true;
   615→   615→    if (Phaser.Input.Keyboard.JustDown(this.keys.K)) this.player.useSkill1();
   616→   616→    if (Phaser.Input.Keyboard.JustDown(this.keys.L)) this.player.useSkill2();
   617→   617→
   618→   618→    this.player.update(dt, move, this.attackQueued);
   619→   619→    this.attackQueued = false;
   620→   620→
   621→   621→    // 적 AI
   622→   622→    for (const e of this.enemies) {
   623→   623→      if (e.active && e.alive) e.tick(dt, this.player);
   624→   624→    }
   625→   625→    // 보스 AI
   626→   626→    this.boss?.tick(dt, this.player);
   627→   627→
   628→   628→    // F2 핵심 2: 화면 가장자리 화살표 — 목표물이 안 보일 때 방향 안내
   629→   629→    this.updateEdgeArrow();
   630→   630→  }

   643→   643→  /* ================= 가독성 (F2) ================= */

   645→   645→  private questTargetPos(): Phaser.Math.Vector2 | null {
   646→   646→    const q = this.currentQuest();
   647→   647→    if (!q) return null;
   648→   648→    switch (q.type) {
   649→   649→      case "collect":
   650→   650→        return this.fragment && this.fragment.active ? new Phaser.Math.Vector2(this.fragment.x, this.fragment.y) : null;
   651→   651→      case "hunt": {
   652→   652→        let best: Enemy | null = null;
   653→   653→        let bd = Infinity;
   654→   654→        for (const e of this.enemies) {
   655→   655→          if (!e.active || !e.alive) continue;
   656→   656→          const d = Phaser.Math.Distance.Between(e.x, e.y, this.player.x, this.player.y);
   657→   657→          if (d < bd) {
   658→   658→            bd = d;
   659→   659→            best = e;
   660→   660→          }
   661→   661→        }
   662→   662→        return best ? new Phaser.Math.Vector2(best.x, best.y) : null;
   663→   663→      }
   664→   664→      case "reach":
   665→   665→        return this.portal && this.portal.active ? new Phaser.Math.Vector2(this.portal.x, this.portal.y) : null;
   666→   666→      case "boss":
   667→   667→        return this.boss && this.boss.active && this.boss.alive
   668→   668→          ? new Phaser.Math.Vector2(this.boss.x, this.boss.y)
   669→   669→          : null;
   670→   670→      default:
   671→   671→        return null;
   672→   672→    }
   673→   673→  }

   675→   675→  private updateEdgeArrow() {
   676→   676→    const target = this.questTargetPos();
   677→   677→    if (!target) {
   678→   678→      if (this.edgeArrow) {
   679→   679→        this.edgeArrow.destroy();
   680→   680→        this.edgeArrow = null;
   681→   681→      }
   682→   682→      if (this.questMark) {
   683→   683→        this.questMark.destroy();
   684→   684→        this.questMark = null;
   685→   685→      }
   686→   686→      return;
   687→   687→    }
   688→   688→
   689→   689→    // 목표물 바로 위 퀘스트 마커(!)
   690→   690→    if (!this.questMark) {
   691→   691→      this.questMark = this.add.image(target.x, target.y - 34, "quest_mark").setDepth(22);
   692→   692→    }
   693→   693→    this.questMark.setPosition(target.x, target.y - 34 + Math.sin(this.time.now / 200) * 3);
   694→   694→
   695→   695→    // 카메라 뷰 밖이면 가장자리 화살표 표시
   696→   696→    const view = this.cameras.main.worldView;
   697→   697→    const margin = 40;
   698→   698→    const inside =
   699→   699→      target.x > view.x + margin &&
   700→   700→      target.x < view.right - margin &&
   701→   701→      target.y > view.y + margin &&
   702→   702→      target.y < view.bottom - margin;
   703→   703→
   704→   704→    if (!inside) {
   705→   705→      if (!this.edgeArrow)
   706→   706→        this.edgeArrow = this.add
   707→   707→          .image(0, 0, "edge_arrow")
   708→   708→          .setDepth(50)
   709→   709→          .setScrollFactor(0)
   710→   710→          .setBlendMode(Phaser.BlendModes.ADD);
   711→   711→      const cx = this.cameras.main.width / 2;
   712→   712→      const cy = this.cameras.main.height / 2;
   713→   713→      const angle = Phaser.Math.Angle.Between(cx, cy, target.x - view.x, target.y - view.y);
   714→   714→      // 화면 중심에서 끝까지 레이캐스트하여 클램프
   715→   715→      const halfW = this.cameras.main.width / 2 - 46;
   716→   716→      const halfH = this.cameras.main.height / 2 - 46;
   717→   717→      const t = Math.min(
   718→   718→        Math.abs(halfW / Math.cos(angle)) || Infinity,
   719→   719→        Math.abs(halfH / Math.sin(angle)) || Infinity
   720→   720→      );
   721→   721→      this.edgeArrow.setPosition(cx + Math.cos(angle) * t, cy + Math.sin(angle) * t);
   722→   722→      this.edgeArrow.setRotation(angle);
   723→   723→      this.edgeArrow.setAlpha(0.65 + Math.sin(this.time.now / 150) * 0.3);
   724→   724→    } else if (this.edgeArrow) {
   725→   725→      this.edgeArrow.destroy();
   726→   726→      this.edgeArrow = null;
   727→   727→    }
   728→   728→  }
