    71→    this.portal = null;
    72→    this.beacon = null;
    73→    this.portalBeacon = null;
    74→    this.questMark = null;

    81→  create() {
    82→    const data = this.registry.get("initData") as {
    83→      stage?: StageKey;
    84→      save?: SaveData;
    85→    };
    86→    const save = data.save;
    87→    const stageKey: StageKey = save ? (save.stage as StageKey) : data.stage ?? "forest";
    88→    this.stageDef = STAGES[stageKey];
    89→    this.stageW = this.stageDef.width;
    90→    this.stageH = this.stageDef.height;
    91→
    92→    /* ---------- 바닥 ---------- */
    93→    const groundTex = stageKey === "forest" ? "tile_grass" : "tile_dark";
    94→    this.add.tileSprite(0, 0, this.stageW, this.stageH, groundTex).setOrigin(0).setDepth(0);
    95→    // 중앙 가로 길
    96→    this.add
    97→      .tileSprite(0, this.stageH / 2 - 52, this.stageW, 104, "tile_path")
    98→      .setOrigin(0)
    99→      .setDepth(0)
   100→      .setAlpha(0.9);
   101→    if (stageKey === "forest") {
   102→      this.add
   103→        .tileSprite(this.stageW * 0.55 - 52, 0, 104, this.stageH, "tile_path")
   104→        .setOrigin(0)
   105→        .setDepth(0)
   106→        .setAlpha(0.85);
   107→    }
   108→
   109→    this.physics.world.setBounds(0, 0, this.stageW, this.stageH);
   110→    this.cameras.main.setBounds(0, 0, this.stageW, this.stageH);
   111→    this.cameras.main.setBackgroundColor(stageKey === "forest" ? "#0a1408" : "#0d0a1e");
   112→
   113→    /* ---------- 장식 (F1: 정의된 소수만) ---------- */
   114→    this.placeDecor(stageKey);
   115→
   116→    /* ---------- 플레이어 ---------- */
   117→    const savedPlayer = save;
   118→    this.player = new Player(this, 180, this.stageH / 2);
   119→    if (savedPlayer) {
   120→      this.player.lv = savedPlayer.lv;
   121→      this.player.atk = savedPlayer.atk;
   122→      this.player.maxHp = savedPlayer.maxHp;
   123→      this.player.hp = this.player.maxHp;
   124→    }
   125→    this.playerRef = this.player;
   126→    this.cameras.main.startFollow(this.player, true, 0.12, 0.12);
   127→    this.physics.add.collider(this.player, this.solidGroup);
   128→
   129→    /* ---------- 적 배치 ---------- */
   130→    const rng = new Phaser.Math.RandomDataGenerator([stageKey]);
   131→    for (const group of this.stageDef.enemies) {
   132→      for (let i = 0; i < group.count; i++) {
   133→        let ex = 0;
   134→        let ey = 0;
   135→        let tries = 0;
   136→        do {
   137→          ex = stageKey === "forest" ? rng.between(this.stageW * 0.42, this.stageW - 140) : rng.between(240, this.stageW - 160);
   138→          ey = rng.between(120, this.stageH - 120);
   139→          tries++;
   140→        } while (Phaser.Math.Distance.Between(ex, ey, this.player.x, this.player.y) < 380 && tries < 30);
   141→        const e = new Enemy(this, ex, ey, group.key);
   142→        this.enemies.push(e);
   143→        this.physics.add.collider(e, this.solidGroup);
   144→      }
   145→    }
   146→
   147→    /* ---------- 퀘스트 오브젝트 ---------- */
   148→    if (stageKey === "forest") {
   149→      this.spawnFragment(this.stageW * 0.78, this.stageH * 0.26);
   150→      this.spawnPortal(this.stageW - 130, this.stageH * 0.52);
   151→    } else {
   152→      this.spawnPortalBossArena();
   153→    }
   154→
   155→    /* ---------- 이펙트 풀 ---------- */
   156→    this.buildFxPools();
   157→
   158→    /* ---------- 입력 ---------- */
   159→    this.setupInput();
   160→
   161→    /* ---------- 사운드/BGM ---------- */
   162→    audio.playBGM(stageKey === "forest" ? "field" : "boss");
   163→
   164→    /* ---------- 오프닝 대사 ---------- */
   165→    this.time.delayedCall(400, () => {
   166→      this.showDialogue(stageKey === "forest" ? "intro" : "alfheimIntro");
   167→    });
   168→
   169→    EventBus.emit("ui:playing");
   170→    this.emitHud();
   171→    this.emitQuest();
   172→
   173→    // F2: 거리 실시간 갱신 (300ms 주기 — 프레임 부담 없음)
   174→    this.questTimer = this.time.addEvent({
   175→      delay: 300,
   176→      loop: true,
   177→      callback: () => {
   178→        this.emitQuest();
   179→        this.emitSkills();
   180→      },
   181→    });
   182→
   183→    this.events.once("shutdown", () => this.cleanup());
   184→  }

   190→  private placeDecor(stageKey: StageKey) {
   191→    const def = this.stageDef;
   192→    const rng = new Phaser.Math.RandomDataGenerator([stageKey + "-decor"]);
   193→    this.solidGroup = this.physics.add.staticGroup();
   194→
   195→    // 나무 & 바위 (충돌 있음)
   196→    for (let i = 0; i < def.treeCount; i++) {
   197→      const x = rng.between(80, this.stageW - 80);
   198→      const y = rng.between(90, this.stageH - 80);
   199→      if (Math.abs(y - this.stageH / 2) < 90) continue; // 길 위엔 안 심음
   200→      const t = this.add.image(x, y, "tree").setDepth(Math.floor(y / 10));
   201→      this.solidGroup.add(t);
   202→      (t.body as Phaser.Physics.Arcade.StaticBody).setSize(24, 16).setOffset(16, 66);
   203→    }
   204→    for (let i = 0; i < def.rockCount; i++) {
   205→      const x = rng.between(80, this.stageW - 80);
   206→      const y = rng.between(90, this.stageH - 80);
   207→      if (Math.abs(y - this.stageH / 2) < 90) continue;
   208→      const r = this.add.image(x, y, "rock").setDepth(Math.floor(y / 10));
   209→      this.solidGroup.add(r);
   210→      (r.body as Phaser.Physics.Arcade.StaticBody).setSize(30, 14).setOffset(3, 10);
   211→    }
   212→
   213→    // F1 핵심: 꽃은 def.flowerCount 송이만 (10 이하)
   214→    const flowers = ["flower_r", "flower_y", "flower_w"];
   215→    for (let i = 0; i < def.flowerCount; i++) {
   216→      const x = rng.between(60, this.stageW - 60);
   217→      const y = rng.between(60, this.stageH - 60);
   218→      this.add.image(x, y, rng.pick(flowers)).setDepth(1).setAlpha(0.95);
   219→    }
   220→  }

   222→  private spawnFragment(x: number, y: number) {
   223→    this.fragment = this.physics.add.sprite(x, y, "fragment").setDepth(4);
   224→    (this.fragment.body as Phaser.Physics.Arcade.Body).setCircle(12, -3, -2);
   225→
   226→    // F2 핵심 1: 하늘까지 닿는 빛 기둥 비컨 — 멀리서도 확실히 보임
   227→    this.beacon = this.add
   228→      .image(x, y - 240, "beacon")
   229→      .setDepth(3)
   230→      .setBlendMode(Phaser.BlendModes.ADD)
   231→      .setAlpha(0.55);
   232→    this.tweens.add({
   233→      targets: this.beacon,
   234→      alpha: { from: 0.4, to: 0.75 },
   235→      scaleX: { from: 1, to: 1.25 },
   236→      duration: 900,
   237→      yoyo: true,
   238→      repeat: -1,
   239→      ease: "Sine.inOut",
   240→    });
   241→    // 바닥 글로우
   242→    const glow = this.add.image(x, y + 8, "glow").setDepth(2).setBlendMode(Phaser.BlendModes.ADD).setScale(1.4);
   243→    this.tweens.add({ targets: glow, scale: 1.7, alpha: 0.7, duration: 800, yoyo: true, repeat: -1 });
   244→    // 뜨는 모션
   245→    this.tweens.add({ targets: this.fragment, y: y - 8, duration: 1100, yoyo: true, repeat: -1, ease: "Sine.inOut" });
   246→
   247→    this.physics.add.overlap(this.player, this.fragment, () => {
   248→      if (!this.fragment || !this.fragment.active) return;
   249→      this.collectFragment(glow);
   250→    });
   251→  }
   252→
   253→  private collectFragment(glow: Phaser.GameObjects.Image) {
   254→    audio.sfx.pickup();
   255→    audio.sfx.questDone();
   256→    this.player.atk += 5; // 파편 보너스
   257→    this.player.healFull();
   258→    glow.destroy();
   259→    this.tweens.add({ targets: this.beacon, alpha: 0, duration: 400, onComplete: () => this.beacon?.destroy() });
   260→    this.beacon = null;
   261→    const f = this.fragment!;
   262→    this.tweens.add({
   263→      targets: f,
   264→      y: f.y - 60,
   265→      alpha: 0,
   266→      scale: 1.6,
   267→      duration: 500,
   268→      onComplete: () => f.destroy(),
   269→    });
   270→    this.fragment = null;
   271→    this.spawnBurstAt(f.x, f.y, 14, 0x9df0ff);
   272→    this.showDialogue("fragment");
   273→    this.advanceQuest();
   274→    this.save();
   275→  }

   277→  private spawnPortal(x: number, y: number) {
   278→    this.portal = this.physics.add.sprite(x, y, "portal0").setDepth(3).setTint(0x777777);
   279→    (this.portal.body as Phaser.Physics.Arcade.Body).setSize(30, 40).setOffset(13, 20);
   280→    this.physics.add.overlap(this.player, this.portal, () => {
   281→      if (!this.portalActive || !this.portal?.active) return;
   282→      this.enterPortal();
   283→    });
   284→  }

   286→  private activatePortal() {
   287→    if (!this.portal) return;
   288→    this.portalActive = true;
   289→    this.portal.clearTint();
   290→    this.portal.play("portal-spin");
   291→    audio.sfx.portal();
   292→    // F2: 차원문에도 작은 비컨
   293→    this.portalBeacon = this.add
   294→      .image(this.portal.x, this.portal.y - 230, "beacon")
   295→      .setDepth(2)
   296→      .setBlendMode(Phaser.BlendModes.ADD)
   297→      .setAlpha(0.4)
   298→      .setTint(0x9d7aff);
   299→    this.tweens.add({
   300→      targets: this.portalBeacon,
   301→      alpha: { from: 0.28, to: 0.5 },
   302→      duration: 800,
   303→      yoyo: true,
   304→      repeat: -1,
   305→    });
   306→    this.showBanner("차원문이 열렸다!");
   307→  }

   309→  private enterPortal() {
   310→    this.portalActive = false;
   311→    audio.sfx.portal();
   312→    this.cameras.main.fadeOut(500, 0, 0, 0);
   313→    this.player.state = "idle";
   314→    this.time.delayedCall(520, () => {
   315→      this.saveStage("alfheim");
   316→      this.scene.restart({ stage: "alfheim" });
   317→    });
   318→  }

   326→  private buildFxPools() {
   327→    // ⚠️ 씬 재시작(스테이지 전환) 시 풀이 누적되지 않도록 리셋
   328→    //   (파괴된 텍스트 재사용 → setText 크래시 원인)
   329→    for (const d of this.dmgPool) d.destroy();
   330→    this.dmgPool = [];
   331→    for (const s of this.slashPool) s.destroy();
   332→    this.slashPool = [];
   333→
   334→    // 공유 파티클 이미터 2종
   335→    this.hitEmitter = this.add.particles(0, 0, "spark", {
   336→      lifespan: 260,
   337→      speed: { min: 60, max: 190 },
   338→      scale: { start: 1, end: 0 },
   339→      emitting: false,
   340→      blendMode: Phaser.BlendModes.ADD,
   341→    }).setDepth(30);
   342→    this.burstEmitter = this.add.particles(0, 0, "spark", {
   343→      lifespan: 480,
   344→      speed: { min: 90, max: 300 },
   345→      scale: { start: 1.4, end: 0 },
   346→      emitting: false,
   347→      blendMode: Phaser.BlendModes.ADD,
   348→    }).setDepth(30);
   349→
   350→    // 데미지 텍스트 12장 고정 풀
   351→    for (let i = 0; i < 12; i++) {
   352→      const t = this.add
   353→        .text(0, 0, "", {
   354→          fontFamily: "sans-serif",
   355→          fontSize: "17px",
   356→          color: "#ffffff",
   357→          stroke: "#1a1020",
   358→          strokeThickness: 4,
   359→          fontStyle: "bold",
   360→        })
   361→        .setDepth(40)
   362→        .setActive(false)
   363→        .setVisible(false);
   364→      this.dmgPool.push(t);
   365→    }
   366→
   367→    // 참격 스프라이트 5장 고정 풀
   368→    for (let i = 0; i < 5; i++) {
   369→      const s = this.add
   370→        .image(0, 0, "slash_arc")
   371→        .setDepth(25)
   372→        .setBlendMode(Phaser.BlendModes.ADD)
   373→        .setActive(false)
   374→        .setVisible(false);
   375→      this.slashPool.push(s);
   376→    }
   377→  }

   379→  spawnDamageText(x: number, y: number, val: number) {
   380→    const t = this.dmgPool.find((d) => d.scene && !d.active);
   381→    if (!t) return; // 풀 소진 시 조용히 포기 (프레임 보호)
   382→    t.setText(`${val}`);
   383→    t.setPosition(x, y)
   384→      .setActive(true)
   385→      .setVisible(true)
   386→      .setAlpha(1)
   387→      .setScale(1);
   388→    this.tweens.add({
   389→      targets: t,
   390→      y: y - 34,
   391→      alpha: 0,
   392→      duration: 550,
   393→      ease: "Quad.out",
   394→      onComplete: () => t.setActive(false).setVisible(false),
   395→    });
   396→  }

   444→  /** F5: 참격 초승달 이펙트 — 풀에서 꺼내 170ms 재생 */
   445→  spawnSlash(x: number, y: number, dir: Phaser.Math.Vector2, alt: boolean, scale = 1) {
   446→    const s = this.slashPool[this.slashIdx];
   447→    this.slashIdx = (this.slashIdx + 1) % this.slashPool.length;
   448→    if (!s || !s.scene) return;
   449→    const rot = dir.x > 0 ? 0 : dir.x < 0 ? Math.PI : dir.y > 0 ? Math.PI / 2 : -Math.PI / 2;
   450→    s.setPosition(x + dir.x * 34, y + dir.y * 34 - 6)
   451→      .setRotation(rot)
   452→      .setFlipY(alt)
   453→      .setActive(true)
   454→      .setVisible(true)
   455→      .setAlpha(1)
   456→      .setScale(0.55 * scale);
   457→    this.tweens.add({
   458→      targets: s,
   459→      scale: 1.15 * scale,
   460→      alpha: 0.15,
   461→      duration: 170,
   462→      ease: "Quad.out",
   463→      onComplete: () => s.setActive(false).setVisible(false),
   464→    });
   465→  }

   469→  /** F5: 명중 시 45ms 전체 정지 + 미세 흔들림 — '베인' 느낌의 핵심 */
   470→  onMeleeConnect(_hits: number) {
   471→    audio.sfx.hit();
   472→    if (!this.hitStopActive) {
   473→      this.hitStopActive = true;
   474→      this.physics.world.pause();
   475→      this.time.delayedCall(45, () => {
   476→        this.physics.world.resume();
   477→        this.hitStopActive = false;
   478→      });
   479→    }
   480→    this.cameras.main.shake(50, 0.003);
   481→  }

   483→  onEnemyKilled(key: "wolf" | "minion", exp: number) {
   484→    // alive 플래그 기준으로 정리 (죽은 개체 즉시 제외)
   485→    this.enemies = this.enemies.filter((e) => e.alive);
   486→    this.totalKills++;
   487→    this.player.gainExp(exp);
   488→    const q = this.currentQuest();
   489→    if (q && q.type === "hunt" && this.stageDef.key === (key === "wolf" ? "forest" : "alfheim")) {
   490→      this.huntCount++;
   491→      if (this.huntCount >= (q.need ?? 0)) {
   492→        audio.sfx.questDone();
   493→        if (this.stageDef.key === "forest") {
   494→          this.showDialogue("wolvesDone");
   495→          this.advanceQuest(); // → 차원문 퀘스트
   496→          this.activatePortal();
   497→        } else {
   498→          this.advanceQuest(); // → 보스 퀘스트
   499→          this.spawnBoss();
   500→        }
   501→        this.save();
   502→      }
   503→    }
   504→    this.emitQuest();
   505→  }

   566→  private setupInput() {
   567→    const kb = this.input.keyboard!;
   568→    this.keys = kb.addKeys(
   569→      "W,A,S,D,UP,DOWN,LEFT,RIGHT,SPACE,J,K,L"
   570→    ) as Record<string, Phaser.Input.Keyboard.Key>;
   571→
   572→    const onMove = (v: { x: number; y: number }) => this.touchMove.set(v.x, v.y);
   573→    const onAtk = () => (this.attackQueued = true);
   574→    const onS1 = () => this.player?.useSkill1();
   575→    const onS2 = () => this.player?.useSkill2();
   576→    const onRespawn = () => this.respawnPlayer();
   577→    const onDialogueDone = () => this.resumeFromDialogue();
   578→
   579→    EventBus.on("input:move", onMove);
   580→    EventBus.on("input:attack", onAtk);
   581→    EventBus.on("input:skill1", onS1);
   582→    EventBus.on("input:skill2", onS2);
   583→    EventBus.on("respawn", onRespawn);
   584→    EventBus.on("dialogue:done", onDialogueDone);
   585→    this.events.once("shutdown", () => {
   586→      EventBus.off("input:move", onMove);
   587→      EventBus.off("input:attack", onAtk);
   588→      EventBus.off("input:skill1", onS1);
   589→      EventBus.off("input:skill2", onS2);
   590→      EventBus.off("respawn", onRespawn);
   591→      EventBus.off("dialogue:done", onDialogueDone);
   592→    });
   593→  }

   595→  update(_time: number, delta: number) {
   596→    const dt = Math.min(delta, 50);
   597→    if (this.dialoguing || !this.player) return;
   598→    if (this.player.state === "dead") return;
   599→
   600→    // 키보드 이동
   601→    const mv = new Phaser.Math.Vector2(0, 0);
   602→    if (this.keys.A.isDown || this.keys.LEFT.isDown) mv.x -= 1;
   603→    if (this.keys.D.isDown || this.keys.RIGHT.isDown) mv.x += 1;
   604→    if (this.keys.W.isDown || this.keys.UP.isDown) mv.y -= 1;
   605→    if (this.keys.S.isDown || this.keys.DOWN.isDown) mv.y += 1;
   606→    if (mv.lengthSq() > 0) mv.normalize();
   607→
   608→    // 터치 우선
   609→    const useTouch = this.touchMove.lengthSq() > 0.01;
   610→    const move = useTouch ? this.touchMove : mv;
   611→
   612→    // 키보드 공격/스킬
   613→    if (Phaser.Input.Keyboard.JustDown(this.keys.SPACE) || Phaser.Input.Keyboard.JustDown(this.keys.J))
   614→      this.attackQueued = true;
   615→    if (Phaser.Input.Keyboard.JustDown(this.keys.K)) this.player.useSkill1();
   616→    if (Phaser.Input.Keyboard.JustDown(this.keys.L)) this.player.useSkill2();
   617→
   618→    this.player.update(dt, move, this.attackQueued);
   619→    this.attackQueued = false;
   620→
   621→    // 적 AI
   622→    for (const e of this.enemies) {
   623→      if (e.active && e.alive) e.tick(dt, this.player);
   624→    }
   625→    // 보스 AI
   626→    this.boss?.tick(dt, this.player);
   627→
   628→    // F2 핵심 2: 화면 가장자리 화살표 — 목표물이 안 보일 때 방향 안내
   629→    this.updateEdgeArrow();
   630→  }

   632→  currentMoveVec() {
   633→    return this.touchMove.lengthSq() > 0.01 ? this.touchMove.clone() : new Phaser.Math.Vector2();
   634→  }

   636→  getAllTargets(): (Enemy | Boss)[] {
   637→    const list: (Enemy | Boss)[] = [];
   638→    for (const e of this.enemies) if (e.active && e.alive) list.push(e);
   639→    if (this.boss && this.boss.active && this.boss.alive) list.push(this.boss);
   640→    return list;
   641→  }

   643→  /* ================= 가독성 (F2) ================= */
   644→
   645→  private questTargetPos(): Phaser.Math.Vector2 | null {
   646→    const q = this.currentQuest();
   647→    if (!q) return null;
   648→    switch (q.type) {
   649→      case "collect":
   650→        return this.fragment && this.fragment.active ? new Phaser.Math.Vector2(this.fragment.x, this.fragment.y) : null;
   651→      case "hunt": {
   652→        let best: Enemy | null = null;
   653→        let bd = Infinity;
   654→        for (const e of this.enemies) {
   655→          if (!e.active || !e.alive) continue;
   656→          const d = Phaser.Math.Distance.Between(e.x, e.y, this.player.x, this.player.y);
   657→          if (d < bd) {
   658→            bd = d;
   659→            best = e;
   660→          }
   661→        }
   662→        return best ? new Phaser.Math.Vector2(best.x, best.y) : null;
   663→      }
   664→      case "reach":
   665→        return this.portal && this.portal.active ? new Phaser.Math.Vector2(this.portal.x, this.portal.y) : null;
   666→      case "boss":
   667→        return this.boss && this.boss.active && this.boss.alive
   668→          ? new Phaser.Math.Vector2(this.boss.x, this.boss.y)
   669→          : null;
   670→      default:
   671→        return null;
   672→    }
   673→  }
   674→
   675→  private updateEdgeArrow() {
   676→    const target = this.questTargetPos();
   677→    if (!target) {
   678→      if (this.edgeArrow) {
   679→        this.edgeArrow.destroy();
   680→        this.edgeArrow = null;
   681→      }
   682→      if (this.questMark) {
   683→        this.questMark.destroy();
   684→        this.questMark = null;
   685→      }
   686→      return;
   687→    }
   688→
   689→    // 목표물 바로 위 퀘스트 마커(!)
   690→    if (!this.questMark) {
   691→      this.questMark = this.add.image(target.x, target.y - 34, "quest_mark").setDepth(22);
   692→    }
   693→    this.questMark.setPosition(target.x, target.y - 34 + Math.sin(this.time.now / 200) * 3);
   694→
   695→    // 카메라 뷰 밖이면 가장자리 화살표 표시
   696→    const view = this.cameras.main.worldView;
   697→    const margin = 40;
   698→    const inside =
   699→      target.x > view.x + margin &&
   700→      target.x < view.right - margin &&
   701→      target.y > view.y + margin &&
   702→      target.y < view.bottom - margin;
   703→
   704→    if (!inside) {
   705→      if (!this.edgeArrow)
   706→        this.edgeArrow = this.add
   707→          .image(0, 0, "edge_arrow")
   708→          .setDepth(50)
   709→          .setScrollFactor(0)
   710→          .setBlendMode(Phaser.BlendModes.ADD);
   711→      const cx = this.cameras.main.width / 2;
   712→      const cy = this.cameras.main.height / 2;
   713→      const angle = Phaser.Math.Angle.Between(cx, cy, target.x - view.x, target.y - view.y);
   714→      // 화면 중심에서 끝까지 레이캐스트하여 클램프
   715→      const halfW = this.cameras.main.width / 2 - 46;
   716→      const halfH = this.cameras.main.height / 2 - 46;
   717→      const t = Math.min(
   718→        Math.abs(halfW / Math.cos(angle)) || Infinity,
   719→        Math.abs(halfH / Math.sin(angle)) || Infinity
   720→      );
   721→      this.edgeArrow.setPosition(cx + Math.cos(angle) * t, cy + Math.sin(angle) * t);
   722→      this.edgeArrow.setRotation(angle);
   723→      this.edgeArrow.setAlpha(0.65 + Math.sin(this.time.now / 150) * 0.3);
   724→    } else if (this.edgeArrow) {
   725→      this.edgeArrow.destroy();
   726→      this.edgeArrow = null;
   727→    }
   728→  }

   736→  private advanceQuest() {
   737→    this.questIdx = Math.min(this.questIdx + 1, this.stageDef.quests.length);

   739→    this.emitQuest();

   742→  emitQuest() {
   743→    const q = this.currentQuest();
   744→    if (!q) {
   745→      EventBus.emit("quest", {
   746→        title: "지역 클리어!",
   747→        desc: "",
   748→        current: 1,
   749→        target: 1,
   750→        distance: null,
   751→      } satisfies QuestState);
   752→      return;
   753→    }
   754→    let current = 0;
   755→    let target = 1;
   756→    if (q.type === "hunt") {
   757→      current = Math.min(this.huntCount, q.need ?? 0);
   758→      target = q.need ?? 0;
   759→    }
   760→    let distance: number | null = null;
   761→    const t = this.questTargetPos();
   762→    if (t) distance = Math.round(Phaser.Math.Distance.Between(this.player.x, this.player.y, t.x, t.y) / 32);
   763→    EventBus.emit("quest", {
   764→      title: q.title,
   765→      desc: q.desc,
   766→      current,
   767→      target,
   768→      distance,
   769→    } satisfies QuestState);
   770→  }
