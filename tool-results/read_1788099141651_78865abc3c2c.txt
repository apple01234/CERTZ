   116→  update(dt: number, move: Phaser.Math.Vector2, attackPressed: boolean) {
   117→    if (this.state === "dead") return;
   118→    const ms = dt;
   119→
   120→    this.atkCooldown = Math.max(0, this.atkCooldown - ms);
   121→    this.skill1Cd = Math.max(0, this.skill1Cd - ms);

   130→      this.mp = Math.min(this.maxMp, this.mp + 5);

   136→      this.dashTime -= ms;

   142→      return;
   143→    }
   144→

   169→        const f = this.facing;

   208→  private doAttack() {
   209→    this.state = "attack";
   210→    this.atkCooldown = 300;
   211→    this.swingDone = false;
   212→    this.slashAlt = !this.slashAlt;
   213→    this.hitSet.clear();
   214→
   215→    // 살짝 돌진 — 정지 상태에서 공격할 때만 (이동 중엔 입력 방향 유지 — 스터터 제거 v2.2)
   216→    const dir = this.aimDir();
   217→    const moving = this.lastMove.lengthSq() > 0.01;
   218→    if (!moving) {
   219→      this.lungeDir.copy(dir);
   220→      this.lungeTime = Player.LUNGE_MS;
   221→      this.setVelocity(dir.x * Player.LUNGE_SPEED, dir.y * Player.LUNGE_SPEED);
   222→    } else {
   223→      this.lungeTime = 0;
   224→      // 이동 관성 유지 — 다음 프레임의 공격 중 이동 분기가 입력 방향으로 이어받음
   225→    }
   226→
   227→    // 실제 방향별 베기 프레임 (측면/위/아래 4프레임 스윙)
   228→    const atkKey = dir.y > 0 ? "hero-atk-down" : dir.y < 0 ? "hero-atk-up" : "hero-atk";
   229→    this.setFlipX(dir.y === 0 && dir.x < 0); // 측면 시트는 오른쪽 기준
   230→    this.play(atkKey);
   231→
   232→    // 베기 타이밍: 65ms 후 판정+참격 스윕 (모션 2프레임 시점)
   233→    this.scene.time.delayedCall(65, () => {
   234→      if (this.state !== "attack") return;
   235→      this.swingDone = true;
   236→      this.scene.spawnSlash(this.x, this.y, dir, this.slashAlt);
   237→      this.scene.sfxSwing();
   238→      // 참격 판정 확대 — 전방 160px x 폭 116px (사용자 지시: 히트박스 크게)
   239→      this.checkMeleeHit(dir, 160, 116, 1.0, 280);
   240→    });
   241→
   242→    this.scene.time.delayedCall(200, () => {
   243→      if (this.state === "attack") {
   244→        this.state = "idle";
   245→        // v2.2 — 이동 입력이 없을 때만 정지 (입력 중이면 다음 프레임 걷기로 자연 연결)
   246→        if (this.lastMove.lengthSq() > 0.01) return;
   247→        this.setVelocity(0, 0);
   248→      }
   249→    });
   250→  }

   252→  /** 조준 방향: 마지막 이동 방향(4방향 스냅) */
   253→  private aimDir(): Phaser.Math.Vector2 {
   254→    const f = this.facing;
   255→    if (Math.abs(f.x) >= Math.abs(f.y)) return new Phaser.Math.Vector2(f.x >= 0 ? 1 : -1, 0);
   256→    return new Phaser.Math.Vector2(0, f.y >= 0 ? 1 : -1);
   257→  }

   259→  /** 광역 근접 판정 — rect + 목표 크기 반영 (보스처럼 큰 적도 실제 몸통에 맞음) */
   260→  checkMeleeHit(
   261→    dir: Phaser.Math.Vector2,
   262→    reach: number,
   263→    width: number,
   264→    dmgMul: number,
   265→    knock: number
   266→  ): number {
   267→    const cx = this.x + dir.x * reach * 0.45;
   268→    const cy = this.y + dir.y * reach * 0.45;
   269→    const halfW = dir.x !== 0 ? reach / 2 : width / 2;
   270→    const halfH = dir.x !== 0 ? width / 2 : reach / 2;
   271→    let hits = 0;
   272→    let anyCrit = false;
   273→    for (const e of this.scene.getAllTargets()) {
   274→      if (!e.active || this.hitSet.has(e)) continue;
   275→      const tw = ((e as unknown as { hitW?: number }).hitW ?? 24) / 2;
   276→      const th = ((e as unknown as { hitH?: number }).hitH ?? 24) / 2;
   277→      const ex = e.x - cx;
   278→      const ey = (e.y - cy) * 1.15; // 조금 관대하게
   279→      if (Math.abs(ex) <= halfW + tw && Math.abs(ey) <= halfH + th) {
   280→        this.hitSet.add(e);
   281→        hits++;
   282→        const { dmg, crit } = this.rollDamage(dmgMul);
   283→        if (crit) {
   284→          this.scene.sfxCrit();
   285→          anyCrit = true;
   286→        }
   287→        e.takeDamage(dmg, dir, knock, crit);
   288→      }
   289→    }
   290→    if (hits > 0) {
   291→      // 크리티컬이 섞인 타격은 crit 등급 — 히트스톱 연장 + 셰이크 강조 (타격감 대비)
   292→      this.scene.onMeleeConnect(hits, anyCrit ? "crit" : "basic");
   293→    }
   294→    return hits;
   295→  }

   300→  useSkill1() {

   313→  /** 전사(+미전직) — 회전베기 360° */
   314→  private skill1Spin() {
   315→    this.scene.sfxSpin();
   316→
   317→    // 회전 방향: 조준 측면 기준 (상하 조준 시 현재 플립 방향 따름)
   318→    const aim = this.aimDir();
   319→    const spin = aim.x !== 0 ? (aim.x > 0 ? 1 : -1) : this.flipX ? -1 : 1;
   320→
   321→    // 360° 궤도 반달 + 충격파 + 스파크
   322→    this.scene.spawnSpinSlash(this.x, this.y, spin);
   323→
   324→    // 몸통(스프라이트) 360° 회전 — 검 뻗은 공격 프레임을 돌려 휘두르는 동작
   325→    this.play("hero-atk");
   326→    this.scene.tweens.add({
   327→      targets: this,
   328→      rotation: spin * Math.PI * 2,
   329→      duration: 250,
   330→      ease: "Cubic.inOut",
   331→      onComplete: () => this.setRotation(0),
   332→    });
   333→
   334→    this.scene.time.delayedCall(90, () => {
   335→      // 360° 전체 판정
   336→      for (const e of this.scene.getAllTargets()) {
   337→        if (!e.active || this.hitSet.has(e)) continue;
   338→        const d = Phaser.Math.Distance.Between(this.x, this.y, e.x, e.y);
   339→        if (d <= 118) {
   340→          this.hitSet.add(e);
   341→          const away = new Phaser.Math.Vector2(e.x - this.x, e.y - this.y).normalize();
   342→          const { dmg, crit } = this.rollDamage(1.6, true);
   343→          if (crit) this.scene.sfxCrit();
   344→          e.takeDamage(dmg, away, 300, crit);
   345→        }
   346→      }
   347→      this.scene.onMeleeConnect(1, "skill");
   348→    });
   349→    this.scene.time.delayedCall(310, () => {
   350→      if (this.state === "attack") this.state = "idle";
   351→    });
   352→    this.scene.emitHud();
   353→  }

   355→  /** 궁수 — 관통 화살 3연발 (부채꼴, 각 화살 최대 2명 관통) */
   356→  private skill1Arrows() {
   357→    const aim = this.aimDir();
   358→    const base = Math.atan2(aim.y, aim.x);
   359→    this.play("hero-atk");
   360→    this.scene.sfxSpin();
   361→    for (let i = 0; i < 3; i++) {
   362→      this.scene.time.delayedCall(i * 110, () => {
   363→        if (this.state === "dead") return;
   364→        const { dmg, crit } = this.rollDamage(1.2, true);
   365→        if (crit) this.scene.sfxCrit();

   765→  buy(key: ItemKey): boolean {
   766→    const item = ITEMS[key];
   767→    if (!item || this.gold < item.price) return false;
   768→    if (item.kind === "consumable") {
   769→      this.gold -= item.price;
   770→      this.addPotion(key === "potion_hp" ? "hp" : "mp");
   771→      return true;
   772→    }
   773→    // BM (v1.9): 버프는 개수 누적, 펫/치장은 1회 구매
   774→    if (item.kind === "buff") {
   775→      this.gold -= item.price;
   776→      this.addBuffItem(key as BuffKey);
   777→      return true;
   778→    }
   779→    if (item.kind === "pet") {
   780→      if (this.pets.includes(key as PetKey)) return false;
   781→      this.gold -= item.price;
   782→      this.pets.push(key as PetKey);
   783→      this.pet = key as PetKey; // 구매 즉시 소환
   784→      return true;
   785→    }
   786→    if (item.kind === "cosmetic") {
   787→      if (this.cosmetics.includes(key as CosmeticKey)) return false;
   788→      this.gold -= item.price;
   789→      this.cosmetics.push(key as CosmeticKey);
   790→      this.cosmetic = key as CosmeticKey; // 구매 즉시 착용
   791→      return true;
   792→    }
   793→    if (this.owned.includes(key)) return false; // 이미 보유한 장비
   794→    this.gold -= item.price;
   795→    this.owned.push(key);
   796→    // 구매 즉시 장착 (더 좋은 티어면 자연스러운 흐름)
   797→    if (item.kind === "weapon") this.weapon = key;
   798→    else if (item.kind === "armor") this.armor = key;
   799→    else if (item.kind === "accessory") this.equip(key);
   800→    return true;
   801→  }

   805→  /** 펫 소환/해제 (가방에서) */

   827→  readonly upMax = UPGRADE_MAX; // v2.0 — +12 (메이플 스타포스식, 구 세이브 +5 호환)
   828→
   829→  /** 다음 강화 비용 — 단계별 눈덩이 곡선 (v2.0 밸런스) */
   830→  upgradeCost(slot: "weapon" | "armor"): number {
   831→    return upgradeCost(slot, this.upgrades[slot]);
   832→  }

   834→  /** 다음 강화 성공률 (%) */
   835→  upgradeRate(slot: "weapon" | "armor"): number {
   836→    return UPGRADE_RATES[this.upgrades[slot]] ?? 0;
   837→  }
   838→
   839→  /** 강화 시도 — 결과 타입 반환 (골드는 성공/실패 모두 소모) */
   840→  tryUpgrade(slot: "weapon" | "armor"): "ok" | "fail" | "max" | "poor" {
   841→    const cur = this.upgrades[slot];
   842→    if (cur >= this.upMax) return "max";
   843→    const cost = this.upgradeCost(slot);
   844→    if (this.gold < cost) return "poor";
   845→    this.gold -= cost;
   846→    const ok = Math.random() * 100 < (UPGRADE_RATES[cur] ?? 0);
   847→    if (ok) {
   848→      this.upgrades[slot] = cur + 1;
   849→      this.scene.sfxUpgradeOk();
   850→      this.scene.spawnPickupText(this.x, this.y - 44, `강화 성공! +${cur + 1}`, "#ffd76a");
   851→    } else if (cur >= UPGRADE_FALLBACK_FROM) {
   852→      // v2.0 — +9 이상 실패 시 1단계 하락 (스타포스식 리스크)
   853→      this.upgrades[slot] = cur - 1;
   854→      this.scene.sfxUpgradeFail();
   855→      this.scene.spawnPickupText(this.x, this.y - 44, `강화 실패… +${cur - 1} 하락`, "#ff9a9a");
   856→    } else {
   857→      this.scene.sfxUpgradeFail();
   858→      this.scene.spawnPickupText(this.x, this.y - 44, "강화 실패…", "#ff9a9a");
   859→    }
   860→    this.scene.emitHud();
   861→    return ok ? "ok" : "fail";
   862→  }
