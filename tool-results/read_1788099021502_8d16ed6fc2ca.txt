   382→/** 챕터별 난이도 배율 (인덱스 = 장번호-2 → forest(2장)부터) */
   383→const CH_HP = [1, 1.2, 1.55, 1.95, 2.45, 3.0, 3.7, 4.5, 5.4];
   384→const CH_ATK = [1, 1.12, 1.28, 1.48, 1.7, 1.95, 2.25, 2.6, 3.0];
   385→const CH_EXP = [1, 1.35, 1.8, 2.35, 3.0, 3.8, 4.8, 6.0, 7.5];
   386→
   387→/** 스테이지 키 → {챕터, 구역} 파싱 */
   388→export function parseStage(key: StageKey): { ch: ChapterKey | "village"; sub: number } {
   389→  if (key === "village" || !key) return { ch: "village", sub: 0 };
   390→  const m = /^(forest|kingdom|alfheim|muspelheim|niflheim|cave|nidavellir|hel|abyss)([1-9]|10)$/.exec(key);
   391→  if (!m) return { ch: "village", sub: 0 };
   392→  return { ch: m[1] as ChapterKey, sub: parseInt(m[2], 10) };
   393→}
   394→
   395→export function chapterSpec(key: StageKey): ChapterSpec | null {
   396→  const { ch } = parseStage(key);
   397→  return CHAPTERS.find((c) => c.key === ch) ?? null;
   398→}

   400→/** 구역별 성장 배율 — 적 HP/ATK/EXP/골드 (사용자 지시 #6/#9 밸런스) */
   401→export function stageScale(key: StageKey): { hp: number; atk: number; exp: number; gold: number } {
   402→  const spec = chapterSpec(key);
   403→  if (!spec) return { hp: 1, atk: 1, exp: 1, gold: 1 };
   404→  const { sub } = parseStage(key);
   405→  const i = spec.num - 2;
   406→  const subMul = 1 + (sub - 1) * 0.055;
   407→  return {
   408→    hp: CH_HP[i] * subMul,
   409→    atk: CH_ATK[i] * (1 + (sub - 1) * 0.045),
   410→    exp: CH_EXP[i] * subMul,
   411→    gold: (1 + i * 0.42) * subMul,
   412→  };
   413→}

   482→function buildStage(spec: ChapterSpec, sub: number): StageDef {
   483→  const key = `${spec.key}${sub}`;
   484→  const prefix = `${spec.key}${sub}`;
   485→  const boss = sub === 10 && !!spec.boss;
   486→  const enemies = spec.enemies.map((g) => ({
   487→    key: g.key,
   488→    count: Math.min(11, g.count + Math.floor(sub / 2)),
   489→  }));
   490→  const def: StageDef = {
   491→    key,
   492→    name: `제${spec.num}장 ${spec.title}`,
   493→    subtitle: `${sub}구역 — ${spec.subtitle}`,
   494→    width: spec.width + sub * 130,
   495→    height: spec.height + sub * 30,
   496→    groundTint: spec.groundTint,
   497→    flowerCount: spec.flowers,
   498→    treeCount: spec.trees,
   499→    rockCount: spec.rocks,
   500→    quests: buildQuests(spec, sub, prefix),
   501→    enemies,
   502→    boss,
   503→    bossKey: boss ? spec.boss : undefined,
   504→    repeat: {
   505→      targetKey: spec.main,
   506→      need: spec.repeat.need + sub,
   507→      gold: Math.round(spec.repeat.gold * stageScale(key).gold * 0.9),
   508→      exp: Math.round(spec.repeat.exp * stageScale(key).exp * 0.9),
   509→      title: spec.repeat.title,
   510→      desc: spec.repeat.desc,
   511→    },
   512→  };
   513→  if (sub === 5) {
   514→    def.elite = {
   515→      key: spec.main,
   516→      hpMult: 7.5,
   517→      atkMult: 1.5,
   518→      name: `${ELITE_TITLE[spec.num - 2]} ${ENEMIES[spec.main].name}`,
   519→    };
   520→  }
   521→  return def;
   522→}

   628→/** 오프닝 대사 매핑 — 구역 1은 챕터 인트로, 나머지는 구역 안내 */
   629→export function stageIntro(key: StageKey): string {
   630→  const spec = chapterSpec(key);
   631→  if (!spec) return "villageIntro";
   632→  const { sub } = parseStage(key);
   633→  if (sub === 1) return spec.intro;
   634→  if (sub === 5) return `eliteWarn${spec.num}`;
   635→  if (sub === 10) return `bossApproach${spec.num}`;
   636→  return `ch${spec.num}Walk${((sub - 2) % 3) + 1}`;
   637→}
