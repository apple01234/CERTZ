   183→    // 런 통계(처치/플레이타임) — 씬 재시작(스테이지 전환)과 무관하게 유지

   195→  create() {
   196→    this.impactFX = new ImpactFX(this);
   197→    const data = this.registry.get("initData") as {
   198→      stage?: StageKey;
   199→      save?: SaveData;
   200→    };
   201→    const save = data.save;
   202→    const stageKey: StageKey = save ? (save.stage as StageKey) : data.stage ?? "forest";
   203→    this.stageDef = STAGES[stageKey];
   204→    this.stageW = this.stageDef.width;
   205→    this.stageH = this.stageDef.height;
   206→
   207→    /* ---------- 바닥 ---------- */
   208→    const groundTex =
   209→      stageKey === "alfheim" ? "tile_dark"
   210→      : stageKey === "cave" ? "tile_cave"
   211→      : stageKey === "abyss" ? "tile_abyss"
   212→      : stageKey === "niflheim" ? "tile_snow"
   213→      : "tile_grass";
   214→    const pathTex =
   215→      stageKey === "niflheim" ? "tile_ice"
   216→      : stageKey === "cave" || stageKey === "abyss" ? "tile_path_dark"
   217→      : "tile_path";
   218→    this.add.tileSprite(0, 0, this.stageW, this.stageH, groundTex).setOrigin(0).setDepth(0);
   219→    // 중앙 가로 길
   220→    this.add
   221→      .tileSprite(0, this.stageH / 2 - 52, this.stageW, 104, pathTex)
   222→      .setOrigin(0)
   223→      .setDepth(0)
   224→      .setAlpha(0.9);
   225→    if (stageKey === "forest") {
   226→      this.add
   227→        .tileSprite(this.stageW * 0.55 - 52, 0, 104, this.stageH, pathTex)
   228→        .setOrigin(0)
   229→        .setDepth(0)
   230→        .setAlpha(0.85);
   231→    }
   232→    // 지형 전환 프린지 + 지면 변형 — 자로 잰 직선 경계/균일 반복 패턴을 자연스럽게 (타일맵 부자연 개선)
   233→    this.buildGroundBlend(stageKey, groundTex, pathTex);
   234→
   235→    this.physics.world.setBounds(0, 0, this.stageW, this.stageH);
   236→    this.cameras.main.setBounds(0, 0, this.stageW, this.stageH);
   237→    this.cameras.main.setBackgroundColor(STAGE_BG[stageKey]);
   238→
   239→    // 반응형: 화면 밀도 유지용 카메라 줌 (RESIZE 캔버스 1:1 + 카메라 확대)
   240→    this.applyCameraZoom();
   241→    this.scale.on("resize", this.applyCameraZoom, this);
   242→
   243→    /* ---------- 장식 (F1: 정의된 소수만) ---------- */
   244→    this.placeDecor(stageKey);
   245→
   246→    /* ---------- 상점 NPC (2D MMORPG 기본 요소) ---------- */
   247→    this.spawnMerchant();
   248→
   249→    /* ---------- 플레이어 ---------- */
   250→    const savedPlayer = save;
   251→    this.player = new Player(this, 180, this.stageH / 2);
   252→    if (savedPlayer) {
   253→      this.player.lv = savedPlayer.lv;
   254→      this.player.atk = savedPlayer.atk;
   255→      this.player.maxHp = savedPlayer.maxHp;
   256→      this.player.hp = this.player.maxHp;
   257→      // RPG 자원 복원 (구 세이브는 loadSave()가 기본값 채움)
   258→      this.player.gold = savedPlayer.gold ?? 30;
   259→      this.player.potions = { hp: savedPlayer.potions?.hp ?? 2, mp: savedPlayer.potions?.mp ?? 1 };
   260→      this.player.weapon = (savedPlayer.weapon ?? "weapon_1") as ItemKey;
   261→      this.player.armor = (savedPlayer.armor ?? "armor_1") as ItemKey;
   262→      this.player.owned = (savedPlayer.owned ?? ["weapon_1", "armor_1"]) as ItemKey[];
   263→      // 강화/장신구 복원 (구 세이브는 loadSave()가 기본값 채움)
   264→      this.player.upgrades.weapon = savedPlayer.upWea ?? 0;
   265→      this.player.upgrades.armor = savedPlayer.upArm ?? 0;
   266→      this.player.accessory = (savedPlayer.accessory ?? null) as ItemKey | null;
   267→      // 퀘스트 진행 복원 (이어하기 — 파편/보상 중복 수령 방지)
   268→      this.savedQuestIdx = { ...(savedPlayer.questIdx ?? {}) };
   269→      this.questIdx = Phaser.Math.Clamp(this.savedQuestIdx[stageKey] ?? 0, 0, this.stageDef.quests.length);
   270→      // 플레이어 이름 복원 (인트로에서 지정)
   271→      if (savedPlayer.playerName) setPlayerName(savedPlayer.playerName);
   272→      // 전직 클래스 복원 (v1.7 — 구 세이브 null 호환)
   273→      this.player.applySavedClass(savedPlayer.cls);
   274→    }
   275→    this.repeatNeed = this.stageDef.repeat?.need ?? 0;
   276→    this.playerRef = this.player;
   277→    this.cameras.main.startFollow(this.player, true, 0.12, 0.12);
   278→    this.physics.add.collider(this.player, this.solidGroup);
   279→
   280→    /* ---------- 적 배치 ---------- */
   281→    const rng = new Phaser.Math.RandomDataGenerator([stageKey]);
   282→    for (const group of this.stageDef.enemies) {
   283→      for (let i = 0; i < group.count; i++) {
   284→        let ex = 0;
   285→        let ey = 0;
   286→        let tries = 0;
   287→        do {
   288→          ex = stageKey === "forest" ? rng.between(this.stageW * 0.42, this.stageW - 140) : rng.between(240, this.stageW - 160);
   289→          ey = rng.between(120, this.stageH - 120);
   290→          tries++;
   291→        } while (Phaser.Math.Distance.Between(ex, ey, this.player.x, this.player.y) < 380 && tries < 30);
   292→        const e = new Enemy(this, ex, ey, group.key);
   293→        this.enemies.push(e);
   294→        this.spawnRecords.push({ key: group.key, x: ex, y: ey });
   295→        this.physics.add.collider(e, this.solidGroup);
   296→      }
   297→    }
   298→
   299→    /* ---------- 퀘스트 오브젝트 ---------- */
   300→    if (stageKey === "village") {
   301→      this.buildVillage();
   302→      // 마을 차원문은 항상 열려 있음 (뿌리숲으로 출발)
   303→      this.spawnPortal(this.stageW - 110, this.stageH * 0.52);
   304→      this.activatePortal(true);
   305→    } else {
   306→      if (!this.stageDef.boss) this.spawnPortal(this.stageW - 130, this.stageH * 0.52);
   307→      // 수확(collect) 퀘스트 진행 중 — 파편 스폰 (이어하기 무결: ATK 중복 수령 방지)
   308→      if (this.currentQuest()?.type === "collect") this.spawnFragmentForQuest();
   309→    }
   310→
   311→    /* ---------- 퀘스트 진행 복구 (이어하기 — 진행 상태 정합, 오브젝트 생성 후) ---------- */
   312→    const bossQuestIdx = this.stageDef.quests.findIndex((q) => q.type === "boss");
   313→    if (this.stageDef.boss) {
   314→      if (this.currentQuest()?.type === "boss") {
   315→        // 보스전 진행 중 세이브 — 입장 직후 보스 등장 (복구 경로는 등장 대사 생략)
   316→        this.time.delayedCall(900, () => {
   317→          if (!this.boss) this.spawnBoss(false);
   318→        });
   319→      } else if (bossQuestIdx >= 0 && this.questIdx > bossQuestIdx && NEXT_STAGE[stageKey]) {
   320→        // 보스 격파 후 세이브 — 차원문 개방 상태 복구 (구 v1.0 클리어 세이브도 이 경로로 계속)
   321→        this.spawnPortal(this.stageW - 130, this.stageH * 0.52);
   322→        this.activatePortal(true);
   323→      }
   324→    } else if (stageKey !== "village" && this.currentQuest()?.type === "reach") {
   325→      // 수확 완료 후 세이브 — 차원문을 열어둔 채 시작 (소프트락 방지)
   326→      this.activatePortal(true);
   327→    }
   328→
   329→    /* ---------- 이펙트 풀 ---------- */
   330→    this.buildFxPools();
   331→
   332→    /* ---------- 미니맵 (2D MMORPG 기본 요소) ---------- */
   333→    this.minimap = this.add.graphics().setDepth(95).setScrollFactor(0).setAlpha(0.85);
   334→    this.redrawMinimap();
   335→
   336→    /* ---------- 입력 ---------- */
   337→    this.setupInput();
   338→
   339→    /* ---------- 사운드/BGM ---------- */
   340→    audio.playBGM(stageKey === "alfheim" || stageKey === "abyss" ? "boss" : "field");
   341→
   342→    /* ---------- 오프닝 대사 (인트로 시퀀스 중이면 인트로가 인계) ---------- */
   343→    if (stageKey === "village" && !savedPlayer?.playerName) {
   344→      // 신규 플레이어 — 책장 넘기기 대신 플레이형 인트로 (이동 → 우물 → 이름 짓기)
   345→      this.startIntroSequence();
   346→    } else {
   347→      this.time.delayedCall(400, () => {
   348→        this.showDialogue(STAGE_INTRO[stageKey]);
   349→      });
   350→    }
   351→
   352→    EventBus.emit("ui:playing");
   353→    this.emitHud();
   354→    this.emitQuest();
   355→    this.emitRpgState();
   356→
   357→    /* ---------- 멀티플레이 (같은 서버 접속자 동기화 — v1.7) ---------- */
   358→    this.initNet();
   359→
   360→    // F2: 거리 실시간 갱신 (300ms 주기 — 프레임 부담 없음) + 미니맵/RPG 상태
   361→    this.questTimer = this.time.addEvent({
   362→      delay: 300,
   363→      loop: true,
   364→      callback: () => {
   365→        this.emitQuest();
   366→        this.emitSkills();
   367→        this.emitRpgState();
   368→        this.redrawMinimap();
   369→      },
   370→    });
   371→
   372→    this.events.once("shutdown", () => this.cleanup());
   373→  }

   380→  }
