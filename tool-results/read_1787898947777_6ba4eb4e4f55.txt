   288→   288→  private placeDecor(stageKey: StageKey) {
   289→   289→    const def = this.stageDef;
   290→   290→    const rng = new Phaser.Math.RandomDataGenerator([stageKey + "-decor"]);
   291→   291→    this.solidGroup = this.physics.add.staticGroup();
   292→   292→
   293→   293→    // 마을 건물/우물/주민 영역 보호 — 장식이 집 위에 심기지 않게
   294→   294→    const vx = def.width / 2;
   295→   295→    const vy = def.height / 2;
   296→   296→    const reserved: [number, number][] =
   297→   297→      stageKey === "village"
   298→   298→        ? [
   299→   299→            [vx - 400, vy - 170],
   300→   300→            [vx + 90, vy - 200],
   301→   301→            [vx - 190, vy + 215],
   302→   302→            [vx, vy],
   303→   303→            [vx + 210, vy + 120],
   304→   304→            [vx - 90, vy - 90],
   305→   305→            [def.width - 110, vy], // 차원문
   306→   306→          ]
   307→   307→        : [];
   308→   308→    const blocked = (x: number, y: number) => reserved.some(([rx, ry]) => Phaser.Math.Distance.Between(x, y, rx, ry) < 170);
   309→   309→
   310→   310→    // 나무 & 소나무 & 바위 (충돌 있음) — 실제 에셋
   311→   311→    for (let i = 0; i < def.treeCount; i++) {
   312→   312→      const x = rng.between(80, this.stageW - 80);
   313→   313→      const y = rng.between(90, this.stageH - 80);
   314→   314→      if (Math.abs(y - this.stageH / 2) < 90) continue; // 길 위엔 안 심음
   315→   315→      if (blocked(x, y)) continue;
   316→   316→      const tex = rng.pick(["tree", "tree", "pine"] as string[]);
   317→   317→      const t = this.add.image(x, y, tex).setDepth(Math.floor(y / 10));
   318→   318→      this.solidGroup.add(t);
   319→   319→      // 64x64 캔버스 하단 줄기 부근만 충돌
   320→   320→      (t.body as Phaser.Physics.Arcade.StaticBody).setSize(20, 14).setOffset(22, 46);
   321→   321→    }
   322→   322→    for (let i = 0; i < def.rockCount; i++) {
   323→   323→      const x = rng.between(80, this.stageW - 80);
   324→   324→      const y = rng.between(90, this.stageH - 80);
   325→   325→      if (Math.abs(y - this.stageH / 2) < 90) continue;
   326→   326→      if (blocked(x, y)) continue;
   327→   327→      const r = this.add.image(x, y, "rock").setDepth(Math.floor(y / 10));
   328→   328→      this.solidGroup.add(r);
   329→   329→      (r.body as Phaser.Physics.Arcade.StaticBody).setSize(44, 20).setOffset(10, 40);
   330→   330→    }
   331→   331→
   332→   332→    // F1 핵심: 꽃은 def.flowerCount 송이만 (10 이하)
   333→   333→    const flowers = ["flower_r", "flower_y", "flower_w"];
   334→   334→    for (let i = 0; i < def.flowerCount; i++) {
   335→   335→      const x = rng.between(60, this.stageW - 60);
   336→   336→      const y = rng.between(60, this.stageH - 60);
   337→   337→      if (blocked(x, y)) continue;
   338→   338→      this.add.image(x, y, rng.pick(flowers)).setDepth(1).setAlpha(0.95);
   339→   339→    }
   340→   340→
   341→   341→    // 심연 구역(알프헤임) 횃불 — 실제 Kenney 횃불 + 온기 글로우
   342→   342→    if (stageKey === "alfheim") {
   343→   343→      for (let i = 0; i < 6; i++) {
   344→   344→        const tx = 200 + (i * (this.stageW - 400)) / 5;
   345→   345→        const ty = i % 2 === 0 ? this.stageH / 2 - 140 : this.stageH / 2 + 140;
   346→   346→        this.add.image(tx, ty, "torch").setDepth(2);
   347→   347→        const g = this.add
   348→   348→          .image(tx, ty, "glow")
   349→   349→          .setDepth(1)
